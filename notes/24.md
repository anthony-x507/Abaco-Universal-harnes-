1. Automated Click Test

Finds all clickable elements (buttons, links) and clicks them one by one, capturing any console errors or page failures.

@router.post("/test/clicks")
async def click_test():
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    results = []
    # Get all clickable elements
    elements = await _page.query_selector_all('a, button, input[type="submit"], [role="button"]')
    for idx, element in enumerate(elements):
        try:
            # Scroll into view and click
            await element.scroll_into_view_if_needed()
            await element.click(timeout=2000)
            # Wait a bit for any navigation
            await asyncio.sleep(0.5)
            # Check for console errors (simplified: just check page still exists)
            title = await _page.title()
            results.append({"index": idx, "selector": await element.evaluate("el => el.outerHTML.slice(0,200)"), "status": "clicked", "title": title})
            # Go back if navigated? For simplicity, we'll not navigate back.
        except Exception as e:
            results.append({"index": idx, "selector": "?", "status": f"error: {str(e)}"})
    return {"results": results}

2. Accessibility Audit (using axe-core via CDP)

We'll inject axe-core and run audit.

@router.get("/audit/accessibility")
async def accessibility_audit():
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    # Inject axe-core from CDN
    await _page.add_script_tag(url="https://cdnjs.cloudflare.com/ajax/libs/axe-core/4.7.2/axe.min.js")
    result = await _page.evaluate("async () => await axe.run()")
    violations = result.get('violations', [])
    return {"violations": violations}

4. Performance Metrics (Lighthouse)

We'll add an endpoint that runs Lighthouse programmatically (requires lighthouse npm package or use Lighthouse CI). For simplicity, we'll use Playwright to collect basic performance metrics via Navigation Timing API.

@router.get("/performance")
async def performance_metrics():
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    metrics = await _page.evaluate("""
        () => {
            const nav = performance.getEntriesByType('navigation')[0];
            return {
                domContentLoaded: nav.domContentLoadedEventEnd,
                loadComplete: nav.loadEventEnd,
                ttfb: nav.responseStart - nav.requestStart,
            };
        }
    """)
    return {"metrics": metrics}

5. Save/Load Session (Cookies)

@router.post("/session/save")
async def save_session(name: str):
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    cookies = await _page.context.cookies()
    with open(f"data/browser_sessions/{name}.json", 'w') as f:
        json.dump(cookies, f)
    return {"status": "saved"}

@router.post("/session/load")
async def load_session(name: str):
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    with open(f"data/browser_sessions/{name}.json", 'r') as f:
        cookies = json.load(f)
    await _page.context.add_cookies(cookies)
    return {"status": "loaded"}

Update index.html to include collapse button in each card (we'll add to chat card and browser card for demonstration). We'll programmatically add handles and collapse buttons in JS.

In app.js at initialization, call:

function makeCardsResizableAndCollapsible() {
    document.querySelectorAll('.card').forEach(card => {
        // Add collapse button if not present
        if (!card.querySelector('.collapse-btn')) {
            const btn = document.createElement('button');
            btn.className = 'collapse-btn';
            btn.innerHTML = '&minus;';
            btn.onclick = () => {
                const body = card.querySelector('div, form, .chat-messages, .browser-results');
                if (body.style.display === 'none') {
                    body.style.display = '';
                    btn.innerHTML = '&minus;';
                } else {
                    body.style.display = 'none';
                    btn.innerHTML = '+';
                }
            };
            card.appendChild(btn);
        }
        // Add resize handle
        if (!card.querySelector('.resizable-handle')) {
            const handle = document.createElement('div');
            handle.className = 'resizable-handle';
            card.appendChild(handle);
            // Simple resize: on mousedown, track movement and adjust width/height
            handle.addEventListener('mousedown', (e) => {
                e.preventDefault();
                const startX = e.clientX;
                const startY = e.clientY;
                const startWidth = card.offsetWidth;
                const startHeight = card.offsetHeight;
                function onMouseMove(e) {
                    card.style.width = startWidth + (e.clientX - startX) + 'px';
                    card.style.height = startHeight + (e.clientY - startY) + 'px';
                }
                function onMouseUp() {
                    document.removeEventListener('mousemove', onMouseMove);
                    document.removeEventListener('mouseup', onMouseUp);
                }
                document.addEventListener('mousemove', onMouseMove);
                document.addEventListener('mouseup', onMouseUp);
            });
        }
    });
}
// Call after DOM ready
makeCardsResizableAndCollapsible();
