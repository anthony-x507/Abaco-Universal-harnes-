📁 plugins/knowledge_base/plugin.py

This plugin uses a vector database (ChromaDB or FAISS) for semantic search over documents. It provides actions:

· ingest_document – load a PDF/Markdown/Text file and index it. 
· search – perform semantic search over indexed documents. 
· list_documents – list all indexed documents. 
· generate_quiz – create a quiz from knowledge base content. 
· get_learning_path – suggest a structured learning path for a topic.

Implementation uses subprocess for document conversion and Python libraries for embeddings.

# plugins/knowledge_base/plugin.py
import subprocess
import json
import logging
from pathlib import Path
from plugins.loader import BasePlugin

class Plugin(BasePlugin):
    """Knowledge base and training plugin."""

    def initialize(self, config):
        self.config = config
        self.vector_db_path = Path(config.get("vector_db_path", "data/knowledge/vector_db"))
        self.document_store = Path(config.get("document_store", "data/knowledge/documents"))
        self.embedding_model = config.get("embedding_model", "all-MiniLM-L6-v2")
        self.vector_db_path.mkdir(parents=True, exist_ok=True)
        self.document_store.mkdir(parents=True, exist_ok=True)
        self.logger = logging.getLogger("knowledge_base")
        self.db = None  # Lazy initialization

    def execute(self, action, parameters):
        if action == "ingest_document":
            return self._ingest_document(parameters)
        elif action == "search":
            return self._search(parameters)
        elif action == "list_documents":
            return self._list_documents()
        elif action == "generate_quiz":
            return self._generate_quiz(parameters)
        elif action == "get_learning_path":
            return self._get_learning_path(parameters)
        else:
            raise NotImplementedError(f"Action '{action}' not supported by knowledge_base")

    def _init_db(self):
        """Initialize ChromaDB client and collection."""
        try:
            import chromadb
            self.client = chromadb.PersistentClient(path=str(self.vector_db_path))
            self.collection = self.client.get_or_create_collection(name="knowledge")
            return True
        except ImportError:
            self.logger.error("chromadb not installed. Run: pip install chromadb")
            return False
        except Exception as e:
            self.logger.error(f"ChromaDB init failed: {e}")
            return False

    def _ingest_document(self, params):
        """
        Ingest a document into the knowledge base.
        params: file_path, metadata (optional)
        """
        file_path = params.get("file_path")
        if not file_path or not Path(file_path).exists():
            return {"error": "file_path not found"}

        # Copy to document store
        source = Path(file_path)
        dest = self.document_store / source.name
        dest.write_bytes(source.read_bytes())

        # Extract text (simplified: assume plain text or use pdftotext if available)
        text = ""
        if source.suffix == ".pdf":
            # Use pdftotext if available
            try:
                result = subprocess.run(["pdftotext", str(source), "-"], capture_output=True, text=True, check=True)
                text = result.stdout
            except FileNotFoundError:
                text = "PDF text extraction requires pdftotext. Install poppler-utils."
        else:
            text = source.read_text(encoding="utf-8", errors="replace")

        # Index with ChromaDB
        if self._init_db():
            import uuid
            doc_id = str(uuid.uuid4())
            self.collection.add(
                documents=[text],
                metadatas=[{"source": source.name, **params.get("metadata", {})}],
                ids=[doc_id]
            )
            return {"status": "ingested", "document_id": doc_id, "characters": len(text)}
        else:
            return {"error": "ChromaDB not available"}

    def _search(self, params):
        """
        Perform semantic search over indexed documents.
        params: query, top_k (default 5)
        """
        query = params.get("query")
        top_k = params.get("top_k", 5)
        if not query:
            return {"error": "query required"}
        if not self._init_db():
            return {"error": "ChromaDB not available"}
        results = self.collection.query(query_texts=[query], n_results=top_k)
        return {"results": results}

    def _list_documents(self):
        """List documents in the document store."""
        files = [f.name for f in self.document_store.iterdir() if f.is_file()]
        return {"documents": files}

    def _generate_quiz(self, params):
        """
        Generate a simple quiz from the knowledge base (placeholder).
        params: topic, num_questions
        """
        topic = params.get("topic", "")
        num_questions = params.get("num_questions", 5)
        # In real implementation, use an LLM to generate questions based on indexed content.
        return {
            "status": "placeholder",
            "topic": topic,
            "num_questions": num_questions,
            "note": "LLM-based quiz generation will be integrated later."
        }

    def _get_learning_path(self, params):
        """
        Suggest a learning path for a given topic (placeholder).
        params: topic
        """
        topic = params.get("topic", "")
        # Could use a predefined mapping or LLM to generate steps.
        return {
            "topic": topic,
            "steps": [
                "Foundational concepts",
                "Core tools and techniques",
                "Hands-on exercises",
                "Advanced topics"
            ]
        }

📁 docs/knowledge_base_training.md

Reference guide for the knowledge base module.

# Knowledge Base & Training Platform Module

This module provides agents with tools to create, manage, and search a technical knowledge base. It also supports generating quizzes and learning paths for training purposes.

## Supported Actions
- `ingest_document` – add a document (PDF, text, markdown) to the knowledge base and index it for semantic search.
- `search` – perform semantic search over indexed documents.
- `list_documents` – list all ingested documents.
- `generate_quiz` – create a quiz from knowledge base content (placeholder).
- `get_learning_path` – suggest a structured learning path for a topic.

## Requirements
- **Python libraries**: `chromadb`, `sentence-transformers` (for embeddings), `pypdf` or `pdftotext` for PDF extraction.
- **Storage**: local vector database at `data/knowledge/vector_db`.
- **Optional**: for large-scale deployment, use a remote vector DB (Weaviate, Pinecone).

## Usage Example
1. Agent loads `knowledge_trainer` template.
2. User asks: "Ingest all my security research PDFs and make them searchable."
3. Agent calls `knowledge_base.ingest_document` for each PDF.
4. User then asks: "Search for Pegasus IOCs."
5. Agent calls `knowledge_base.search` with the query and returns relevant documents.
6. For training, user asks: "Create a quiz on reverse engineering basics."
7. Agent (future) uses LLM to generate quiz from indexed content.

## Notes
- The current implementation is a scaffold; actual embedding and LLM-based quiz generation require additional setup.
- The plugin can be extended to support web pages, GitHub repos, and APIs as sources.

## Extending
- Integrate with `LangChain` or `LlamaIndex` for advanced retrieval-augmented generation (RAG).
- Add support for multiple collections and namespaces.
- Implement automatic document summarization before indexing.
