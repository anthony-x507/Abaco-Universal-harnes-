export interface Agent {
  agent_id: string;
  name: string;
  status: string;
  folder: string;
  spec: Record<string, any>;
}

export interface AgentOptions {
  providers: string[];
  channels: string[];
  templates: string[];
  plugins: string[];
}

export interface AgentCreateSpec {
  template?: string;
  name?: string;
  llm_provider?: string;
  llm_model?: string;
  plugins?: string[];
  communication?: string[];
  languages?: string[];
  system_prompt?: string;
  memory?: boolean;
  sandbox?: boolean;
  autostart?: boolean;
}

export async function listAgents(): Promise<Agent[]> {
  return fetchJSON<Agent[]>('/agents');
}

export async function createAgent(spec: AgentCreateSpec): Promise<Agent> {
  return fetchJSON<Agent>('/agents', {
    method: 'POST',
    body: JSON.stringify(spec),
  });
}

export async function startAgent(agentId: string): Promise<{ status: string }> {
  return fetchJSON<{ status: string }>(`/agents/${agentId}/start`, { method: 'POST' });
}

export async function stopAgent(agentId: string): Promise<{ status: string }> {
  return fetchJSON<{ status: string }>(`/agents/${agentId}/stop`, { method: 'POST' });
}

export async function deleteAgent(agentId: string): Promise<{ status: string }> {
  return fetchJSON<{ status: string }>(`/agents/${agentId}`, { method: 'DELETE' });
}

export async function deployAgent(agentId: string, repoName?: string): Promise<{ clone_url: string }> {
  const query = repoName ? `?repo_name=${encodeURIComponent(repoName)}` : '';
  return fetchJSON<{ clone_url: string }>(`/agents/${agentId}/deploy${query}`, { method: 'POST' });
}

export async function getAgentOptions(): Promise<AgentOptions> {
  return fetchJSON<AgentOptions>('/agents/options');
}

import { useState, useEffect } from 'react';
import { getAgentOptions } from '../lib/api';
import { AgentCreateSpec } from '../lib/api';
import { AgentOptions } from '../lib/api';

interface Props {
  onCreate: (spec: AgentCreateSpec) => void;
  onCancel: () => void;
}

export default function AgentCreateForm({ onCreate, onCancel }: Props) {
  const [options, setOptions] = useState<AgentOptions | null>(null);
  const [spec, setSpec] = useState<AgentCreateSpec>({
    template: '',
    name: '',
    llm_provider: '',
    llm_model: '',
    plugins: [],
    communication: [],
    languages: ['en'],
    memory: true,
    sandbox: false,
    autostart: false,
  });

  useEffect(() => {
    (async () => {
      try {
        const opts = await getAgentOptions();
        setOptions(opts);
        if (opts.providers.length > 0) {
          setSpec((prev) => ({ ...prev, llm_provider: opts.providers[0] }));
        }
      } catch (e) {
        console.error(e);
      }
    })();
  }, []);

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    onCreate(spec);
  };

  if (!options) return <div>Loading options...</div>;

  return (
    <div className="bg-surface1 border border-border rounded-lg p-4 mb-4">
      <h2 className="text-lg font-medium mb-4">Create New Agent</h2>
      <form onSubmit={handleSubmit} className="space-y-3">
        <div>
          <label className="block text-sm text-gray-400">Template</label>
          <select
            value={spec.template}
            onChange={(e) => setSpec({ ...spec, template: e.target.value })}
            className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
          >
            <option value="">Custom</option>
            {options.templates.map((tpl) => (
              <option key={tpl} value={tpl}>
                {tpl}
              </option>
            ))}
          </select>
        </div>
        <div>
          <label className="block text-sm text-gray-400">Name</label>
          <input
            type="text"
            value={spec.name}
            onChange={(e) => setSpec({ ...spec, name: e.target.value })}
            placeholder="Agent name (optional)"
            className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
          />
        </div>
        <div className="grid grid-cols-2 gap-3">
          <div>
            <label className="block text-sm text-gray-400">LLM Provider</label>
            <select
              value={spec.llm_provider}
              onChange={(e) => setSpec({ ...spec, llm_provider: e.target.value })}
              className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
            >
              {options.providers.map((p) => (
                <option key={p} value={p}>
                  {p}
                </option>
              ))}
            </select>
          </div>
          <div>
            <label className="block text-sm text-gray-400">Model</label>
            <input
              type="text"
              value={spec.llm_model}
              onChange={(e) => setSpec({ ...spec, llm_model: e.target.value })}
              placeholder="Override model"
              className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
            />
          </div>
        </div>
        <div>
          <label className="block text-sm text-gray-400">Plugins (comma separated)</label>
          <input
            type="text"
            value={(spec.plugins || []).join(',')}
            onChange={(e) =>
              setSpec({
                ...spec,
                plugins: e.target.value.split(',').map((s) => s.trim()).filter(Boolean),
              })
            }
            className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
          />
        </div>
        <div>
          <label className="block text-sm text-gray-400">Communication Channels (comma separated)</label>
          <input
            type="text"
            value={(spec.communication || []).join(',')}
            onChange={(e) =>
              setSpec({
                ...spec,
                communication: e.target.value.split(',').map((s) => s.trim()).filter(Boolean),
              })
            }
            className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
          />

        <div>
          <label className="block text-sm text-gray-400">Languages (comma separated)</label>

import { useState, useEffect } from 'react';
import AgentList from '../components/AgentList';
import AgentCreateForm from '../components/AgentCreateForm';
import { listAgents, createAgent, startAgent, stopAgent, deleteAgent, deployAgent, Agent, AgentCreateSpec } from '../lib/api';

export default function AgentsPage() {
  const [agents, setAgents] = useState<Agent[]>([]);
  const [showCreate, setShowCreate] = useState(false);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadAgents();
  }, []);

  const loadAgents = async () => {
    setLoading(true);
    try {
      const data = await listAgents();
      setAgents(data);
    } catch (e) {
      console.error(e);
    } finally {
      setLoading(false);
    }
  };

  const handleCreate = async (spec: AgentCreateSpec) => {
    try {
      await createAgent(spec);
      setShowCreate(false);
      loadAgents();
    } catch (e) {
      console.error(e);
      alert('Failed to create agent: ' + (e as Error).message);
    }
  };

  const handleStart = async (id: string) => {
    try {
      await startAgent(id);
      loadAgents();
    } catch (e) {
      alert('Failed to start agent: ' + (e as Error).message);
    }
  };

  const handleStop = async (id: string) => {
    try {
      await stopAgent(id);
      loadAgents();
    } catch (e) {
      alert('Failed to stop agent: ' + (e as Error).message);
    }
  };

  const handleDelete = async (id: string) => {
    if (!confirm('Delete this agent?')) return;
    try {
      await deleteAgent(id);
      loadAgents();
    } catch (e) {
      alert('Failed to delete agent: ' + (e as Error).message);
    }
  };

  const handleDeploy = async (id: string) => {
    const repoName = prompt('Enter GitHub repo name (optional):') || undefined;
    try {
      const result = await deployAgent(id, repoName);
      alert('Deployed! Clone URL: ' + result.clone_url);
    } catch (e) {
      alert('Failed to deploy agent: ' + (e as Error).message);
    }
  };

  return (
    <div className="p-4">
      <div className="flex justify-between items-center mb-4">
        <h1 className="text-xl font-semibold">Agents</h1>
        <button
          onClick={() => setShowCreate(!showCreate)}
          className="bg-accent text-black font-medium rounded-lg px-4 py-2 hover:opacity-90"
        >
          {showCreate ? 'Close Form' : 'Create Agent'}
        </button>
      </div>

      {showCreate && <AgentCreateForm onCreate={handleCreate} onCancel={() => setShowCreate(false)} />}

      {loading ? (
        <div className="text-gray-400">Loading agents...</div>
      ) : (
        <AgentList
          agents={agents}
          onStart={handleStart}
          onStop={handleStop}
          onDelete={handleDelete}
          onDeploy={handleDeploy}
        />
      )}
    </div>
  );
}
