📁 plugins/embedded_dev/plugin.py

This plugin uses command‑line tools like arduino-cli, platformio, avrdude, openocd, yosys, nextpnr, and iverilog. It provides actions:

· list_boards – detect connected boards. 
· compile_code – compile source for a target board. 
· flash_firmware – flash compiled binary to device. 
· serial_monitor – read/write serial data. 
· synthesize_fpga – synthesize Verilog/VHDL to bitstream. 
· program_fpga – upload bitstream to FPGA board.

Implementation uses subprocess and returns command results.

# plugins/embedded_dev/plugin.py
import subprocess
import json
import logging
from pathlib import Path
from plugins.loader import BasePlugin

class Plugin(BasePlugin):
    """Embedded systems and FPGA development plugin."""

    def initialize(self, config):
        self.config = config
        self.arduino_cli = config.get("arduino_cli_path", "arduino-cli")
        self.platformio = config.get("platformio_path", "platformio")
        self.fpga_toolchain = config.get("fpga_toolchain", "yosys")
        self.workspace = Path(config.get("workspace_dir", "data/embedded_workspace"))
        self.workspace.mkdir(parents=True, exist_ok=True)
        self.logger = logging.getLogger("embedded_dev")

    def execute(self, action, parameters):
        if action == "list_boards":
            return self._list_boards()
        elif action == "compile_code":
            return self._compile_code(parameters)
        elif action == "flash_firmware":
            return self._flash_firmware(parameters)
        elif action == "serial_monitor":
            return self._serial_monitor(parameters)
        elif action == "synthesize_fpga":
            return self._synthesize_fpga(parameters)
        elif action == "program_fpga":
            return self._program_fpga(parameters)
        else:
            raise NotImplementedError(f"Action '{action}' not supported by embedded_dev")

    def _list_boards(self):
        """List connected Arduino/ESP boards using arduino-cli or platformio."""
        try:
            result = subprocess.run([self.arduino_cli, "board", "list"], capture_output=True, text=True, timeout=10)
            return {"boards": result.stdout}
        except FileNotFoundError:
            return {"error": "arduino-cli not found."}
        except Exception as e:
            return {"error": str(e)}

    def _compile_code(self, params):
        """
        Compile code for a target board.
        params: project_dir, board_fqbn (e.g., arduino:avr:uno), sketch_name
        """
        project_dir = params.get("project_dir")
        board_fqbn = params.get("board_fqbn", "arduino:avr:uno")
        if not project_dir:
            return {"error": "project_dir required"}
        cmd = [self.arduino_cli, "compile", "--fqbn", board_fqbn, project_dir]
        result = subprocess.run(cmd, capture_output=True, text=True)
        return {"status": "success" if result.returncode == 0 else "failed",
                "output": result.stdout + result.stderr}

    def _flash_firmware(self, params):
        """
        Flash compiled firmware to a connected board.
        params: project_dir, board_fqbn, port
        """
        project_dir = params.get("project_dir")
        board_fqbn = params.get("board_fqbn", "arduino:avr:uno")
        port = params.get("port", "/dev/ttyACM0")
        cmd = [self.arduino_cli, "upload", "--fqbn", board_fqbn, "-p", port, project_dir]
        result = subprocess.run(cmd, capture_output=True, text=True)
        return {"status": "success" if result.returncode == 0 else "failed",
                "output": result.stdout + result.stderr}

    def _serial_monitor(self, params):
        """
        Open serial monitor (read/write). For simplicity, we return command to run.
        params: port, baudrate
        """
        port = params.get("port", "/dev/ttyACM0")
        baud = params.get("baudrate", 115200)
        return {"command": f"screen {port} {baud}"}

    def _synthesize_fpga(self, params):
        """
        Synthesize Verilog/VHDL using yosys and nextpnr.
        params: source_files (list), top_module, fpga_part (e.g., ice40)
        """
        sources = params.get("source_files", [])
        top = params.get("top_module", "top")
        part = params.get("fpga_part", "ice40")
        # Placeholder: create a simple command sequence
        commands = [
            f"yosys -p 'read_verilog {' '.join(sources)}; synth_ice40 -top {top}; write_json design.json'",
            f"nextpnr-ice40 --json design.json --pcf pins.pcf --asc design.asc",
            "icepack design.asc design.bin"
        ]
        return {"commands": commands, "note": "Run these in order in the project directory."}

    def _program_fpga(self, params):
        """
        Upload bitstream to FPGA board (e.g., ice40 using iceprog).
        params: bitstream_file, programmer
        """
        bitstream = params.get("bitstream_file", "design.bin")
        programmer = params.get("programmer", "iceprog")
        cmd = [programmer, bitstream]
        result = subprocess.run(cmd, capture_output=True, text=True)
        return {"status": "success" if result.returncode == 0 else "failed",
                "output": result.stdout + result.stderr}

# Embedded Systems & FPGA Development Module

This module provides agents with tools for developing firmware for microcontrollers and programming FPGAs. It supports Arduino, ESP32, STM32, and open‑source FPGA toolchains (IceStorm, Yosys).

## Supported Actions
- `list_boards` – list connected development boards.
- `compile_code` – compile firmware using arduino-cli.
- `flash_firmware` – upload firmware to a board.
- `serial_monitor` – open serial monitor (command).
- `synthesize_fpga` – synthesize Verilog/VHDL to bitstream.
- `program_fpga` – program an FPGA board.

## Requirements
- **Host tools** (or sandbox): `arduino-cli`, `platformio`, `avrdude`, `openocd`, `yosys`, `nextpnr-ice40`, `iverilog`.
- **Hardware**: Arduino/ESP32/STM32 board, or Lattice iCE40 FPGA board for FPGA work.
- **Permissions**: USB/serial access.

## Usage Example
1. Agent loads `embedded_engineer` template.
2. User says: "Write a program to blink an LED on Arduino Uno and flash it."
3. Agent uses `code_tools` to generate the sketch.
4. Agent calls `embedded_dev.compile_code` with the project directory.
5. Agent calls `embedded_dev.flash_firmware` to upload.
6. User can monitor serial output via `serial_monitor`.

## Notes
- FPGA synthesis requires a `.pcf` pin constraint file for the target board.
- For STM32, consider using PlatformIO with the appropriate platform.
- The sandbox image includes most tools, but real hardware access may require `--privileged` mode.

## Extending
- Add support for RISC‑V development (e.g., SiFive, GD32V).
- Integrate ESP-IDF for ESP32 advanced features.
- Add VHDL synthesis flow (using GHDL + nextpnr).
- Implement automated hardware testing scripts.
