1. Update Scheduler to Execute Skills

We'll modify api/routes/scheduler.py to perform the replay via an internal HTTP call to our own /v1/browser/record/replay endpoint. This keeps the code decoupled and ensures the browser automation is reused.

Updated api/routes/scheduler.py

# api/routes/scheduler.py
import asyncio
import json
import time
import httpx
from pathlib import Path
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from datetime import datetime, timedelta

router = APIRouter()

scheduled_tasks = []

class ScheduleRequest(BaseModel):
    skill_name: str
    run_at: str  # ISO datetime or "now"
    repeat_interval_minutes: int = 0  # 0 = run once

@router.post("/scheduler/add")
async def add_schedule(request: ScheduleRequest):
    task = {
        "skill_name": request.skill_name,
        "run_at": request.run_at,
        "repeat_interval_minutes": request.repeat_interval_minutes,
        "status": "scheduled",
        "last_run": None,
        "error": None,
    }
    scheduled_tasks.append(task)
    # Start background loop if not already running
    asyncio.create_task(_scheduler_loop())
    return {"status": "scheduled", "task": task}

async def _scheduler_loop():
    while True:
        now = datetime.utcnow()
        for task in scheduled_tasks:
            if task["status"] not in ("scheduled", "retry"):
                continue
            run_at = datetime.fromisoformat(task["run_at"])
            if now >= run_at:
                # Execute the skill via internal API call to browser replay
                try:
                    async with httpx.AsyncClient() as client:
                        response = await client.post(
                            f"http://localhost:8000/v1/browser/record/replay",
                            params={"skill_name": task["skill_name"]},
                            timeout=60,
                        )
                        if response.status_code == 200:
                            task["status"] = "executed"
                            task["last_run"] = now.isoformat()
                            task["error"] = None
                        else:
                            task["status"] = "error"
                            task["error"] = f"Replay failed with status {response.status_code}"
                except Exception as e:
                    task["status"] = "error"
                    task["error"] = str(e)

                # If repeat_interval_minutes > 0, schedule next run
                if task["repeat_interval_minutes"] > 0:
                    next_run = now + timedelta(minutes=task["repeat_interval_minutes"])
                    task["run_at"] = next_run.isoformat()
                    task["status"] = "scheduled"
        await asyncio.sleep(10)  # check every 10 seconds

@router.get("/scheduler/list")
async def list_schedules():
    return {"tasks": scheduled_tasks}

@router.delete("/scheduler/{task_index}")
async def delete_schedule(task_index: int):
    if 0 <= task_index < len(scheduled_tasks):
        removed = scheduled_tasks.pop(task_index)
        return {"removed": removed}
    raise HTTPException(status_code=404, detail="Task not found")

async function loadScheduledTasks() {
    try {
        const data = await fetchJSON(`${API_BASE}/scheduler/list`);
        const container = document.getElementById('scheduled-tasks-list');
        if (data.tasks.length === 0) {
            container.innerHTML = '<p>No tasks scheduled.</p>';
            return;
        }
        container.innerHTML = '';
        data.tasks.forEach((task, index) => {
            const item = document.createElement('div');
            item.className = 'scheduled-task-item';
            item.innerHTML = `
                <span><strong>${task.skill_name}</strong> - ${task.status}</span>
                <small>Run at: ${task.run_at}</small>
                ${task.error ? `<div class="error">${task.error}</div>` : ''}
                <button onclick="deleteScheduledTask(${index})">Delete</button>
            `;
            container.appendChild(item);
        });
    } catch (e) {
        console.error('Failed to load scheduled tasks', e);
    }
}

async function deleteScheduledTask(index) {
    try {
        await fetchJSON(`${API_BASE}/scheduler/${index}`, { method: 'DELETE' });
        loadScheduledTasks();
    } catch (e) { alert('Delete failed: ' + e.message); }
}

document.getElementById('refresh-scheduled-btn').addEventListener('click', loadScheduledTasks);
