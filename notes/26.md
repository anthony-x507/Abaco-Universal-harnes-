        raise HTTPException(status_code=400, detail="No active tab")
    _recording = True

        window.__recordedActions = [];
        window.__generateSelector = (el) => {
            if (el.id) return '#' + el.id;
            if (el.name) return `[name="${el.name}"]`;
            let path = [];
            while (el.nodeType === Node.ELEMENT_NODE) {
                let selector = el.nodeName.toLowerCase();
                if (el.className) {
                    selector += '.' + Array.from(el.classList).join('.');
                }
                let nth = 1;
                let sibling = el;
                while (sibling = sibling.previousElementSibling) {
                    if (sibling.nodeName.toLowerCase() === selector) nth++;
                }
                if (nth > 1) selector += `:nth-of-type(${nth})`;
                path.unshift(selector);
                el = el.parentNode;
                if (el === document.body) break;
            }
            return path.join(' > ');
        };
        document.addEventListener('click', (e) => {
            if (window.__recordedActions) {
                const selector = window.__generateSelector(e.target);
                window.__recordedActions.push({type:'click', selector: selector});
            }
        }, true);
        document.addEventListener('input', (e) => {
            if (window.__recordedActions && e.target.tagName === 'INPUT') {
                const selector = window.__generateSelector(e.target);
                window.__recordedActions.push({type:'input', selector: selector, value: e.target.value});
            }
        }, true);
        document.addEventListener('change', (e) => {
            if (window.__recordedActions && e.target.tagName === 'SELECT') {

4. Replay a Skill

@router.post("/record/replay")
async def record_replay(skill_name: str):
    page = get_active_page()
    if not page:
        raise HTTPException(status_code=400, detail="No active tab")
    skills_dir = Path("data/browser_skills")
    file_path = skills_dir / f"{skill_name}.json"
    if not file_path.exists():
        raise HTTPException(status_code=404, detail="Skill not found")
    with open(file_path, 'r') as f:
        actions = json.load(f)
    results = []
    for action in actions:
        try:
            if action['type'] == 'click':
                await page.click(action['selector'], timeout=2000)
            elif action['type'] == 'input':
                await page.fill(action['selector'], action.get('value', ''))
            elif action['type'] == 'select':
                await page.select_option(action['selector'], action.get('value', ''))
            results.append({"status": "ok", "action": action})
        except Exception as e:
            results.append({"status": "error", "action": action, "error": str(e)})
    return {"results": results}
