# api/routes/factory.py (updated)
from fastapi import APIRouter, HTTPException
from ..schemas import FactoryStatusResponse, CodeExecuteRequest
from ..main import factory_instance
from sandbox.linux_sandbox import LinuxSandbox
from factory.options import get_available_providers, get_available_channels
from integrations.llm_providers import PROVIDER_REGISTRY

router = APIRouter()

# Initialize sandbox (if docker available)
try:
    sandbox = LinuxSandbox(ttl_seconds=300, image="python:3.10-slim")
except Exception as e:
    sandbox = None
    print(f"Sandbox initialization failed: {e}")

@router.get("/models")
async def list_models():
    """List supported LLM providers and their default models."""
    providers = get_available_providers()
    # Build a simple dict with provider name and its default model if defined
    result = []
    for name in providers:
        provider_class = PROVIDER_REGISTRY.get(name)
        default_model = getattr(provider_class, 'default_model', '')
        result.append({"provider": name, "default_model": default_model})
    return {"providers": result}

@router.get("/communication")
async def list_communication_channels():
    """List available communication channels."""
    channels = get_available_channels()
    return {"channels": channels}

@router.post("/code/execute")
async def execute_code(request: CodeExecuteRequest):
    """Execute code in a sandbox and return output."""
    if not sandbox:
        raise HTTPException(status_code=500, detail="Sandbox unavailable (Docker not running?)")
    container_id = sandbox.create_sandbox(bot_id="api_code_executor")
    try:
        # Write code to a temp file inside container to avoid escaping issues.
        import tempfile
        with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
            f.write(request.code)
            tmpfile = f.name
        # Copy file into container
        import subprocess
        subprocess.run(["docker", "cp", tmpfile, f"{container_id}:/tmp/code.py"], check=True)
        # Execute
        output = sandbox.execute(container_id, "python /tmp/code.py")
        # Cleanup temp file
        import os
        os.unlink(tmpfile)
        return {"output": output}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        sandbox.remove_sandbox(container_id)

Note: We added a more robust code execution using docker cp and a temp file. This avoids shell escaping issues.
