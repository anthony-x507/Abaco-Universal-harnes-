# communication/base.py
"""
Base class for all communication channels.
Every channel must implement start(), stop(), send_message().
"""

import logging
from typing import Optional, Dict, Any, Callable


class BaseCommunication:
    """Abstract base for communication channels."""

    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.logger = logging.getLogger(self.__class__.__name__)
        self.is_running = False

    def start(self):
        """Connect and start listening for incoming messages (if applicable)."""
        self.is_running = True
        self.logger.info(f"{self.__class__.__name__} started.")

    def stop(self):
        """Disconnect and stop listening."""
        self.is_running = False
        self.logger.info(f"{self.__class__.__name__} stopped.")

    def send_message(self, recipient: str, text: str) -> bool:
        """Send a message to a recipient."""
        raise NotImplementedError

    def set_incoming_callback(self, callback: Callable[[str, str], None]):
        """
        Set a callback to handle incoming messages.
        callback(sender, message)
        """
        self.incoming_callback = callback

    def _handle_incoming(self, sender: str, message: str):
        """Called internally when a message arrives."""
        if hasattr(self, 'incoming_callback') and self.incoming_callback:
            self.incoming_callback(sender, message)

# communication/telegram.py
"""
Telegram channel using python-telegram-bot.
Requires TELEGRAM_BOT_TOKEN in environment or config.
"""

import asyncio
import logging
from typing import Optional, Dict, Any
from .base import BaseCommunication

try:
    from telegram import Update
    from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes
    TELEGRAM_AVAILABLE = True
except ImportError:
    TELEGRAM_AVAILABLE = False


class TelegramChannel(BaseCommunication):
    """Telegram bot channel."""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.bot_token = config.get("bot_token") or __import__('os').environ.get("TELEGRAM_BOT_TOKEN")
        if not self.bot_token:
            raise ValueError("Telegram bot token not provided.")
        if not TELEGRAM_AVAILABLE:
            raise ImportError("python-telegram-bot is not installed. Run: pip install python-telegram-bot")
        self.application = None

    async def _start(self):
        self.application = Application.builder().token(self.bot_token).build()
        # Add handlers
        self.application.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, self._message_handler))
        await self.application.initialize()
        await self.application.start()
        await self.application.updater.start_polling()

    def _message_handler(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        sender = str(update.effective_user.id)
        message = update.message.text
        self._handle_incoming(sender, message)

    def start(self):
        """Start the Telegram bot (blocking? No, we run in async)."""
        self.is_running = True
        # Use asyncio to run in a separate thread or integrate with main loop
        import threading
        self._thread = threading.Thread(target=asyncio.run, args=(self._start(),))
        self._thread.daemon = True
        self._thread.start()
        self.logger.info("Telegram channel started.")

    def stop(self):
        if self.application:
            # We can't cleanly stop async loop from another thread easily; for now just mark stopped.
            self.is_running = False
            self.logger.info("Telegram channel stopped (may need manual cleanup).")

    def send_message(self, recipient: str, text: str) -> bool:
        """Send a message to a Telegram user by ID."""
        import asyncio
        async def send():
            await self.application.bot.send_message(chat_id=recipient, text=text)
        try:
            asyncio.run_coroutine_threadsafe(send(), self.application.loop)
            return True
        except Exception as e:
            self.logger.error(f"Failed to send Telegram message: {e}")
            return False

# communication/discord.py
"""
Discord channel using discord.py.
Requires DISCORD_BOT_TOKEN in environment or config.
"""

import asyncio
import logging
from typing import Optional, Dict, Any
from .base import BaseCommunication

try:
    import discord
    DISCORD_AVAILABLE = True
except ImportError:
    DISCORD_AVAILABLE = False


class DiscordChannel(BaseCommunication):
    """Discord bot channel."""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.bot_token = config.get("bot_token") or __import__('os').environ.get("DISCORD_BOT_TOKEN")
        if not self.bot_token:
            raise ValueError("Discord bot token not provided.")
        if not DISCORD_AVAILABLE:
            raise ImportError("discord.py is not installed. Run: pip install discord.py")
        intents = discord.Intents.default()
        intents.message_content = True
        self.client = discord.Client(intents=intents)
        self.client.event(self.on_ready)
        self.client.event(self.on_message)

    async def on_ready(self):
        self.logger.info(f"Discord bot logged in as {self.client.user}")

    async def on_message(self, message):
        if message.author == self.client.user:
            return
        self._handle_incoming(str(message.author.id), message.content)

    def start(self):
        self.is_running = True
        import threading
        self._thread = threading.Thread(target=self.client.run, args=(self.bot_token,))
        self._thread.daemon = True
        self._thread.start()
        self.logger.info("Discord channel started.")

    def stop(self):
        self.is_running = False
        # Discord client is not easily stoppable from another thread; we'll just mark.
        self.logger.info("Discord channel stopped (may need manual cleanup).")

    def send_message(self, recipient: str, text: str) -> bool:
        # For Discord, recipient can be user ID or channel ID.
        # We'll assume channel ID for simplicity.
        import asyncio
        async def send():
            channel = self.client.get_channel(int(recipient))
            if channel:
                await channel.send(text)
        try:
            asyncio.run_coroutine_threadsafe(send(), self.client.loop)
            return True
        except Exception as e:
            self.logger.error(f"Failed to send Discord message: {e}")
            return False

# communication/slack.py
"""
Slack channel using slack_sdk.
Requires SLACK_BOT_TOKEN and SLACK_APP_TOKEN (for Socket Mode) in environment or config.
"""

import logging
from typing import Optional, Dict, Any
from .base import BaseCommunication

try:
    from slack_sdk import WebClient
    from slack_sdk.socket_mode import SocketModeClient
    from slack_sdk.socket_mode.request import SocketModeRequest
    from slack_sdk.socket_mode.response import SocketModeResponse
    SLACK_AVAILABLE = True
except ImportError:
    SLACK_AVAILABLE = False


class SlackChannel(BaseCommunication):
    """Slack bot channel using Socket Mode."""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.bot_token = config.get("bot_token") or __import__('os').environ.get("SLACK_BOT_TOKEN")
        self.app_token = config.get("app_token") or __import__('os').environ.get("SLACK_APP_TOKEN")
        if not self.bot_token or not self.app_token:
            raise ValueError("Slack bot token and app token required.")
        if not SLACK_AVAILABLE:
            raise ImportError("slack_sdk is not installed. Run: pip install slack_sdk")
        self.web_client = WebClient(token=self.bot_token)
        self.socket_client = SocketModeClient(
            app_token=self.app_token,
            web_client=self.web_client
        )
        self.socket_client.socket_mode_request_listeners.append(self._handle_request)

    def _handle_request(self, client: SocketModeClient, req: SocketModeRequest):
        if req.type == "events_api":
            event = req.payload.get("event", {})
            if event.get("type") == "message" and "text" in event:
                sender = event.get("user", "unknown")
                text = event.get("text", "")
                self._handle_incoming(sender, text)
            # Acknowledge
            response = SocketModeResponse(envelope_id=req.envelope_id)
            client.send_socket_mode_response(response)

    def start(self):
        self.is_running = True
        import threading
        self._thread = threading.Thread(target=self.socket_client.connect)
        self._thread.daemon = True
        self._thread.start()
        self.logger.info("Slack channel started.")

    def stop(self):
        self.is_running = False
        self.socket_client.close()
        self.logger.info("Slack channel stopped.")

    def send_message(self, recipient: str, text: str) -> bool:
        # recipient is channel ID or user ID
        try:
            self.web_client.chat_postMessage(channel=recipient, text=text)
            return True
        except Exception as e:
            self.logger.error(f"Failed to send Slack message: {e}")
            return False

# communication/email_smtp.py
"""
Email channel using SMTP.
Requires SMTP_USERNAME and SMTP_PASSWORD in environment or config.
"""

import smtplib
import logging
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from typing import Optional, Dict, Any
from .base import BaseCommunication


class EmailSMTPChannel(BaseCommunication):
    """Send emails via SMTP."""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.smtp_server = config.get("smtp_server", "smtp.gmail.com")
        self.smtp_port = config.get("smtp_port", 587)
        self.username = config.get("username") or __import__('os').environ.get("SMTP_USERNAME")
        self.password = config.get("password") or __import__('os').environ.get("SMTP_PASSWORD")
        if not self.username or not self.password:
            raise ValueError("SMTP credentials not provided.")

    def send_message(self, recipient: str, text: str) -> bool:
        try:
            msg = MIMEMultipart()
            msg['From'] = self.username
            msg['To'] = recipient
            msg['Subject'] = "Message from Aegis Agent"
            msg.attach(MIMEText(text, 'plain'))
            server = smtplib.SMTP(self.smtp_server, self.smtp_port)
            server.starttls()
            server.login(self.username, self.password)
            server.sendmail(self.username, recipient, msg.as_string())
            server.quit()
            return True
        except Exception as e:
            self.logger.error(f"Failed to send email: {e}")
            return False

# communication/webhook.py
"""
Generic webhook channel.
Sends HTTP POST requests to a specified URL.
"""

import requests
import logging
from typing import Optional, Dict, Any
from .base import BaseCommunication


class WebhookChannel(BaseCommunication):
    """Send messages via outgoing webhook."""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.webhook_url = config.get("webhook_url")
        if not self.webhook_url:
            raise ValueError("Webhook URL is required.")

    def send_message(self, recipient: str, text: str) -> bool:
        try:
            payload = {
                "recipient": recipient,
                "text": text
            }
            response = requests.post(self.webhook_url, json=payload, timeout=10)
            return response.status_code == 200
        except Exception as e:
            self.logger.error(f"Failed to send webhook: {e}")
            return False

# communication/manager.py
"""
Communication Manager - manages multiple channel instances for an agent.
"""

import logging
from typing import Dict, Any, List, Optional
from .base import BaseCommunication
from .telegram import TelegramChannel
from .discord import DiscordChannel
from .slack import SlackChannel
from .email_smtp import EmailSMTPChannel
from .webhook import WebhookChannel


class CommunicationManager:
    """Holds and manages communication channels for a single agent."""

    def __init__(self, channel_configs: Dict[str, Dict[str, Any]]):
        self.channels: Dict[str, BaseCommunication] = {}
        self.logger = logging.getLogger("comm_manager")
        self._load_channels(channel_configs)

    def _load_channels(self, configs: Dict[str, Dict[str, Any]]):
        channel_classes = {
            "telegram": TelegramChannel,
            "discord": DiscordChannel,
            "slack": SlackChannel,
            "email_smtp": EmailSMTPChannel,
            "webhook": WebhookChannel,
            # Add more as implemented
        }
        for name, config in configs.items():
            if name in channel_classes:
                try:
                    channel = channel_classes[name](config)
                    self.channels[name] = channel
                    self.logger.info(f"Loaded communication channel: {name}")
                except Exception as e:
                    self.logger.error(f"Failed to load channel {name}: {e}")
            else:
                self.logger.warning(f"Unknown channel type: {name}")

    def start_all(self):
        for name, channel in self.channels.items():
            channel.start()

    def stop_all(self):
        for name, channel in self.channels.items():
            channel.stop()

    def send_message(self, channel_name: str, recipient: str, text: str) -> bool:
        if channel_name in self.channels:
            return self.channels[channel_name].send_message(recipient, text)
        return False

    def set_incoming_callback(self, callback):
        for channel in self.channels.values():
            channel.set_incoming_callback(callback)

🔄 Integration with Agent Factory

Each agent created by the Factory can have a communication section in its config (as seen in templates). When the agent starts, the Factory can create a CommunicationManager with those channel configs and connect it to the agent's message processing pipeline. Incoming messages from any channel are forwarded to the agent as user messages; agent responses can be sent back through the same channel.

This is the foundation for 30+ communication channels — we've implemented 5, and adding more follows the same pattern.
