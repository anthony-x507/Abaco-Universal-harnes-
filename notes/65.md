1. universal/plugins/terminal.py

```python
import subprocess
from universal.plugins import Plugin

class TerminalPlugin(Plugin):
    name = "terminal_access"
    description = "Execute shell commands"

    def tools(self):
        return {
            "execute_command": {
                "description": "Execute a shell command",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "command": {"type": "string", "description": "Command to execute"}
                    },
                    "required": ["command"]
                },
                "handler": self._execute
            }
        }

    def _execute(self, command: str, **kwargs) -> str:
        try:
            result = subprocess.run(command, shell=True, capture_output=True, text=True, timeout=30)
            output = result.stdout
            if result.stderr:
                output += f"\n[stderr]\n{result.stderr}"
            return output.strip() or "[no output]"
        except subprocess.TimeoutExpired:
            return "Error: command timed out"
        except Exception as e:
            return f"Error: {e}"
```

import subprocess
import os
from universal.plugins import Plugin

class TTSPlugin(Plugin):
    name = "tts"
    description = "Text-to-Speech with voice and speed control"

    def tools(self):
        return {
            "speak": {
                "description": "Convert text to speech",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "text": {"type": "string", "description": "Text to speak"},
                        "voice": {
                            "type": "string",
                            "enum": ["male", "female"],
                            "description": "Voice type"
                        },
                        "speed": {
                            "type": "number",
                            "enum": [0.5, 0.8, 1.0, 1.2, 1.5, 2.0],
                            "description": "Speaking speed"
                        }
                    },
                    "required": ["text"]
                },
                "handler": self._speak
            }
        }

    def _speak(self, text: str, voice: str = "male", speed: float = 1.0, **kwargs) -> str:
        try:
            if os.name == "posix":
                voice_name = "Alex" if voice == "male" else "Victoria"
                speed_param = f"-r {int(speed * 175)}"
                cmd = f"say -v {voice_name} {speed_param} '{text.replace("'", "'\\''")}'"
                result = os.system(cmd)
                return f"Speaking: {text}" if result == 0 else "Error: say command failed"
            elif os.name == "nt":
                import win32com.client
                speaker = win32com.client.Dispatch("SAPI.SpVoice")
                speaker.Voice = speaker.GetVoices().Item(0 if voice == "male" else 1)
                speaker.Rate = int((speed - 1) * 10)
                speaker.Speak(text)
                return f"Speaking: {text}"
            else:
                return "Error: TTS not supported"
        except Exception as e:
            return f"Error: {e}"
```

import whisper
import tempfile
import os
from universal.plugins import Plugin

class STTPlugin(Plugin):
    name = "stt"
    description = "Speech-to-Text using Whisper"

    def tools(self):
        return {
            "transcribe": {
                "description": "Transcribe audio file to text",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "audio_path": {"type": "string", "description": "Path to audio file"}
                    },
                    "required": ["audio_path"]
                },
                "handler": self._transcribe
            }
        }

    def __init__(self):
        self.model = None

    def _transcribe(self, audio_path: str, **kwargs) -> str:
        try:
            if not os.path.exists(audio_path):
                return f"Error: file not found: {audio_path}"
            if self.model is None:
                self.model = whisper.load_model("base")
            result = self.model.transcribe(audio_path)
            return result["text"].strip()
        except Exception as e:
            return f"Error: {e}"

import base64
from universal.plugins import Plugin

class VisionPlugin(Plugin):
    name = "vision"
    description = "Analyze images using vision-capable LLM"

    def tools(self):
        return {
            "describe_image": {
                "description": "Describe an image",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "image_path": {"type": "string", "description": "Path to image"},
                        "prompt": {
                            "type": "string",
                            "description": "Optional prompt",
                            "default": "Describe this image"
                        }
                    },
                    "required": ["image_path"]
                },
                "handler": self._describe
            }
        }

    def _describe(self, image_path: str, prompt: str = "Describe this image", **kwargs) -> str:
        try:
            from universal.providers import get_current_provider
            provider = get_current_provider()
            if not provider:
                return "Error: No provider available"
            with open(image_path, "rb") as f:
                img_b64 = base64.b64encode(f.read()).decode("utf-8")
            messages = [{"role": "user", "content": [
                {"type": "text", "text": prompt},
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{img_b64}"}}
            ]}]
            return provider.complete(messages)
        except Exception as e:
            return f"Error: {e}"
```

5. universal/plugins/web_search.py

```python
import requests
from universal.plugins import Plugin

class WebSearchPlugin(Plugin):
    name = "web_search"
    description = "Search online using DuckDuckGo"

    def tools(self):
        return {
            "search_web": {
                "description": "Search the web",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "query": {"type": "string", "description": "Search query"},
                        "max_results": {
                            "type": "integer",
                            "description": "Max results",
                            "default": 5
                        }
                    },
                    "required": ["query"]
                },
                "handler": self._search
            }
        }

    def _search(self, query: str, max_results: int = 5, **kwargs) -> str:
        try:
            url = "https://api.duckduckgo.com/"
            params = {"q": query, "format": "json", "no_html": 1, "skip_disambig": 1}
            resp = requests.get(url, params=params, timeout=10)
            data = resp.json()
            results = []
            if data.get("AbstractText"):
                results.append(f"[Abstract] {data['AbstractText']}")
            for topic in data.get("RelatedTopics", [])[:max_results]:
                if "Text" in topic:
                    results.append(f"- {topic['Text']}")
            return "\n".join(results) or "No results found."
        except Exception as e:
            return f"Error: {e}"
```

6. universal/plugins/scraper.py

```python
import requests
from bs4 import BeautifulSoup
from universal.plugins import Plugin

class ScraperPlugin(Plugin):
    name = "scraper"
    description = "Scrape web pages"

    def tools(self):
        return {
            "scrape_url": {
                "description": "Scrape a URL",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "url": {"type": "string", "description": "URL to scrape"},
                        "max_length": {
                            "type": "integer",
                            "description": "Max characters",
                            "default": 5000
                        }
                    },
                    "required": ["url"]
                },
                "handler": self._scrape
            }
        }

    def _scrape(self, url: str, max_length: int = 5000, **kwargs) -> str:
        try:
            headers = {"User-Agent": "Mozilla/5.0"}
            resp = requests.get(url, headers=headers, timeout=15)
            resp.raise_for_status()
            soup = BeautifulSoup(resp.text, "html.parser")
            for tag in soup(["script", "style"]):
                tag.decompose()
            text = soup.get_text(separator="\n", strip=True)
            lines = [line.strip() for line in text.splitlines() if line.strip()]
            text = "\n".join(lines)
            return text[:max_length] + ("\n... [truncated]" if len(text) > max_length else "")
        except Exception as e:
            return f"Error: {e}"
```
