1. Update src/components/Sidebar.tsx to be resizable

We'll add a drag handle on the right edge of the sidebar and use local state to adjust width.

import { useState, useRef, useEffect, useCallback } from 'react';
import { Link, useLocation } from 'react-router-dom';
import { MessageSquare, Bot, Globe, Puzzle, Calendar, Settings } from 'lucide-react';

const navItems = [
  { path: '/', label: 'Chat', icon: MessageSquare },
  { path: '/agents', label: 'Agents', icon: Bot },
  { path: '/browser', label: 'Browser', icon: Globe },
  { path: '/plugins', label: 'Plugins', icon: Puzzle },
  { path: '/scheduler', label: 'Scheduler', icon: Calendar },
  { path: '/settings', label: 'Settings', icon: Settings },
];

export default function Sidebar() {
  const location = useLocation();
  const [width, setWidth] = useState(80); // initial width px
  const sidebarRef = useRef<HTMLElement>(null);
  const isResizing = useRef(false);

  const startResize = useCallback((e: React.MouseEvent) => {
    e.preventDefault();
    isResizing.current = true;
    document.addEventListener('mousemove', handleMouseMove);
    document.addEventListener('mouseup', stopResize);
  }, []);

  const handleMouseMove = useCallback((e: MouseEvent) => {
    if (!isResizing.current) return;
    const newWidth = e.clientX;
    if (newWidth >= 60 && newWidth <= 200) {
      setWidth(newWidth);
    }
  }, []);

  const stopResize = useCallback(() => {
    isResizing.current = false;
    document.removeEventListener('mousemove', handleMouseMove);
    document.removeEventListener('mouseup', stopResize);
  }, []);

  useEffect(() => {
    return () => {
      document.removeEventListener('mousemove', handleMouseMove);
      document.removeEventListener('mouseup', stopResize);
    };
  }, []);

  return (
    <aside
      ref={sidebarRef}
      className="relative bg-surface1 border-r border-border flex flex-col items-center py-4 shrink-0"
      style={{ width: `${width}px` }}
    >
      <div className="text-accent text-2xl font-bold mb-6">A</div>
      <nav className="flex flex-col gap-4">
        {navItems.map(({ path, label, icon: Icon }) => (
          <Link
            key={path}
            to={path}
            className={`p-2 rounded-lg hover:bg-surface2 ${location.pathname === path ? 'text-accent' : 'text-gray-400'}`}
            title={label}
          >
            <Icon size={20} />
          </Link>
        ))}
      </nav>
      {/* Resize handle */}
      <div
        onMouseDown={startResize}
        className="absolute right-0 top-0 h-full w-1 cursor-col-resize hover:bg-accent/50"
      />
    </aside>
  );
}

2. Create src/components/CommandPalette.tsx

A simple modal that appears when pressing Ctrl+K. It lists quick actions and navigates.

import { useEffect, useState, useRef } from 'react';
import { useNavigate } from 'react-router-dom';

interface Command {
  label: string;
  shortcut?: string;
  action: () => void;
}

export default function CommandPalette({ onClose }: { onClose: () => void }) {
  const navigate = useNavigate();
  const [query, setQuery] = useState('');
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    inputRef.current?.focus();
  }, []);

  const commands: Command[] = [
    { label: 'Go to Chat', action: () => navigate('/') },
    { label: 'Go to Agents', action: () => navigate('/agents') },
    { label: 'Go to Browser', action: () => navigate('/browser') },
    { label: 'Go to Plugins', action: () => navigate('/plugins') },
    { label: 'Go to Scheduler', action: () => navigate('/scheduler') },
    { label: 'Go to Settings', action: () => navigate('/settings') },
    { label: 'New Chat', action: () => { navigate('/'); /* optionally trigger new chat */ } },
    { label: 'Close', action: onClose },
  ];

  const filtered = commands.filter((cmd) =>
    cmd.label.toLowerCase().includes(query.toLowerCase())
  );

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Escape') onClose();
    if (e.key === 'Enter' && filtered.length > 0) {
      filtered[0].action();
      onClose();
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-start justify-center pt-20 bg-black/60" onClick={onClose}>
      <div
        className="bg-surface1 border border-border rounded-lg shadow-xl w-full max-w-lg overflow-hidden"
        onClick={(e) => e.stopPropagation()}
      >
        <input
          ref={inputRef}
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onKeyDown={handleKeyDown}
          placeholder="Type a command..."
          className="w-full bg-surface2 border-b border-border px-4 py-3 text-sm focus:outline-none"
        />
        <div className="max-h-80 overflow-y-auto">
          {filtered.map((cmd, idx) => (
            <div
              key={idx}
              className={`px-4 py-2 text-sm cursor-pointer ${
                idx === 0 ? 'bg-surface2 text-accent' : 'hover:bg-surface2'
              }`}
              onClick={() => {
                cmd.action();
                onClose();
              }}
            >
              {cmd.label}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

4. Update src/App.tsx to integrate command palette and activity state

We'll manage isCommandPaletteOpen and listen for Ctrl+K. Also, we'll create a simple global activity context or just use local state for now. For demonstration, we'll pass isBusy false; but later we can wire it to actual API calls using a context. For now, we'll set it to false and note future improvement.

import { useState, useEffect } from 'react';
import { BrowserRouter as Router, Routes, Route } from 'react-router-dom';
import Sidebar from './components/Sidebar';
import TopBar from './components/TopBar';
import CommandPalette from './components/CommandPalette';
import ChatPage from './pages/ChatPage';
import AgentsPage from './pages/AgentsPage';
import BrowserPage from './pages/BrowserPage';
import PluginsPage from './pages/PluginsPage';
import SchedulerPage from './pages/SchedulerPage';
import SettingsPage from './pages/SettingsPage';

function App() {
  const [isCommandPaletteOpen, setCommandPaletteOpen] = useState(false);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.ctrlKey && e.key === 'k') {
        e.preventDefault();
        setCommandPaletteOpen((prev) => !prev);
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  return (
    <Router>
      <div className="flex h-screen">
        <Sidebar />
        <div className="flex flex-col flex-1">
          <TopBar onOpenCommandPalette={() => setCommandPaletteOpen(true)} />
          <main className="flex-1 overflow-auto">
            <Routes>
              <Route path="/" element={<ChatPage />} />
              <Route path="/agents" element={<AgentsPage />} />
              <Route path="/browser" element={<BrowserPage />} />
              <Route path="/plugins" element={<PluginsPage />} />
              <Route path="/scheduler" element={<SchedulerPage />} />
              <Route path="/settings" element={<SettingsPage />} />
            </Routes>
          </main>
        </div>
      </div>
      {isCommandPaletteOpen && (
        <CommandPalette onClose={() => setCommandPaletteOpen(false)} />
      )}
    </Router>
  );
}

export default App;
