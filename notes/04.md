# integrations/llm_providers.py
"""
Unified LLM Provider Manager.
Supports 40+ providers through a common interface.
Each provider can be selected by name when creating an agent.
"""

import os
import logging
from typing import Dict, Any, List, Optional


class BaseLLMProvider:
    """Base class for all LLM providers."""

    api_key_env: str = ""
    default_model: str = ""

    def __init__(self, api_key: Optional[str] = None, model: Optional[str] = None):
        self.api_key = api_key or os.getenv(self.api_key_env, "")
        self.model = model or self.default_model
        self.logger = logging.getLogger(self.__class__.__name__)

    def generate(self, messages: List[Dict[str, str]], **kwargs) -> str:
        raise NotImplementedError

    def stream(self, messages: List[Dict[str, str]], **kwargs):
        raise NotImplementedError


class OpenAIProvider(BaseLLMProvider):
    api_key_env = "OPENAI_API_KEY"
    default_model = "gpt-4o-mini"

    def generate(self, messages, **kwargs):
        import openai
        openai.api_key = self.api_key
        response = openai.ChatCompletion.create(
            model=self.model,
            messages=messages,
            **kwargs
        )
        return response.choices[0].message.content.strip()


class AnthropicProvider(BaseLLMProvider):
    api_key_env = "ANTHROPIC_API_KEY"
    default_model = "claude-3-5-sonnet-20241022"

    def generate(self, messages, **kwargs):
        import anthropic
        client = anthropic.Anthropic(api_key=self.api_key)
        system = ""
        if messages and messages[0]["role"] == "system":
            system = messages[0]["content"]
            messages = messages[1:]
        response = client.messages.create(
            model=self.model,
            system=system,
            messages=messages,
            **kwargs
        )
        return response.content[0].text.strip()


class GoogleProvider(BaseLLMProvider):
    api_key_env = "GOOGLE_API_KEY"
    default_model = "gemini-1.5-pro"

    def generate(self, messages, **kwargs):
        import google.generativeai as genai
        genai.configure(api_key=self.api_key)
        model = genai.GenerativeModel(self.model)
        prompt = "\n".join([f"{m['role']}: {m['content']}" for m in messages])
        response = model.generate_content(prompt)
        return response.text.strip()

class OllamaProvider(BaseLLMProvider):
    api_key_env = ""
    default_model = "llama3.1"

    def generate(self, messages, **kwargs):
        import requests
        prompt = "\n".join([f"{m['role']}: {m['content']}" for m in messages])
        response = requests.post(
            "http://localhost:11434/api/generate",
            json={"model": self.model, "prompt": prompt, "stream": False},
            timeout=120
        )
        data = response.json()
        return data.get("response", "").strip()

# Add more providers here as needed.
# Each provider is a class with a generate() method.
# The registry maps user-friendly names to classes.

PROVIDER_REGISTRY: Dict[str, BaseLLMProvider] = {
    "openai": OpenAIProvider,
    "anthropic": AnthropicProvider,
    "google": GoogleProvider,
    "deepseek": DeepSeekProvider,
    "cohere": CohereProvider,
    "mistral": MistralProvider,
    "ollama": OllamaProvider,
    "groq": GroqProvider,
    # Add 32+ more providers here
}


def get_llm(provider_name: str, api_key: Optional[str] = None, model: Optional[str] = None) -> BaseLLMProvider:
    """Factory function to get an LLM provider instance."""
    provider_class = PROVIDER_REGISTRY.get(provider_name.lower())
    if not provider_class:
        raise ValueError(f"Unknown provider: {provider_name}")
    return provider_class(api_key=api_key, model=model)


def list_supported_providers() -> List[str]:
    """Return list of supported provider names."""
    return list(PROVIDER_REGISTRY.keys())

📄 How to Add More Providers

To add a new provider (for example, Together AI, Replicate, Fireworks, etc.), you simply:

1. Create a new class inheriting from BaseLLMProvider.
2. Set api_key_env and default_model.
3. Implement generate().
4. Add the class to PROVIDER_REGISTRY.

Example:

class TogetherProvider(BaseLLMProvider):
    api_key_env = "TOGETHER_API_KEY"
    default_model = "mistralai/Mixtral-8x7B-Instruct-v0.1"

    def generate(self, messages, **kwargs):
        import openai
        openai.api_key = self.api_key
        openai.api_base = "https://api.together.xyz/v1"
        response = openai.ChatCompletion.create(
            model=self.model,
            messages=messages,
            **kwargs
        )
        return response.choices[0].message.content.strip()

PROVIDER_REGISTRY["together"] = TogetherProvider

That's it. The rest of the system doesn't change.

🔄 Integration with Agent Factory

When creating an agent via the Factory, the user selects a provider name (e.g., "deepseek") and optionally a model. The Factory stores this in the agent's config, and when the agent needs to call an LLM, it uses:

provider = get_llm(config["default_llm"], model=config.get("default_model"))
response = provider.generate(messages)

This makes the Factory model-agnostic and ready for 40+ LLMs.
