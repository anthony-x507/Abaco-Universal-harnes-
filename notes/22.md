const API_BASE = 'http://localhost:8000/v1';

let activeRequests = 0;
let currentConversationId = null;

function showActivity() {
    activeRequests++;
    document.getElementById('activity-indicator').style.display = 'flex';
}

function hideActivity() {
    activeRequests--;
    if (activeRequests <= 0) {
        activeRequests = 0;
        document.getElementById('activity-indicator').style.display = 'none';
    }
}

async function fetchJSON(url, options = {}) {
    showActivity();
    try {
        const response = await fetch(url, {
            headers: {
                'Content-Type': 'application/json',
            },
            ...options,
        });
        if (!response.ok) {
            const error = await response.text();
            throw new Error(error);
        }
        return response.json();
    } finally {
        hideActivity();
    }
}

// ---------- Agent Management Functions ----------

async function loadOptions() {
    try {
        const options = await fetchJSON(`${API_BASE}/agents/options`);
        populateSelect('provider', options.providers);
    } catch (err) {
        document.getElementById('create-message').innerText = 'Failed to load options: ' + err.message;
    }
}

function populateSelect(elementId, items) {
    const select = document.getElementById(elementId);
    select.innerHTML = '';
    items.forEach(item => {
        const option = document.createElement('option');
        option.value = item;
        option.textContent = item;
        select.appendChild(option);
    });
}

async function loadStatus() {
    try {
        const status = await fetchJSON(`${API_BASE}/status`);
        document.getElementById('factory-status').innerHTML = `
            <strong>Agents:</strong> ${status.agents} &nbsp;|&nbsp;
            <strong>Running:</strong> ${status.running_agents.join(', ') || 'None'} &nbsp;|&nbsp;
            <strong>Plugins:</strong> ${status.plugins.join(', ') || 'None'}
        `;
    } catch (err) {
        document.getElementById('factory-status').innerText = 'Failed to load status: ' + err.message;
    }
}

async function loadAgents() {
    try {
        const agents = await fetchJSON(`${API_BASE}/agents`);
        const container = document.getElementById('agent-list');
        if (agents.length === 0) {
            container.innerHTML = '<p>No agents created yet.</p>';
            return;
        }
        container.innerHTML = '';
        agents.forEach(agent => {
            const item = document.createElement('div');
            item.className = 'agent-item';
            item.innerHTML = `
                <div class="agent-info">
                    <strong>${agent.name}</strong> <small>(${agent.agent_id})</small> - <em>${agent.status}</em>
                </div>
                <div class="agent-actions">
                    <button onclick="startAgent('${agent.agent_id}')">Start</button>
                    <button onclick="stopAgent('${agent.agent_id}')">Stop</button>
                    <button onclick="deleteAgent('${agent.agent_id}')">Delete</button>
                    <button onclick="deployAgent('${agent.agent_id}')">Deploy</button>
                </div>
            `;
            container.appendChild(item);
        });
    } catch (err) {
        document.getElementById('agent-list').innerText = 'Failed to load agents: ' + err.message;
    }
}

async function createAgent(event) {
    event.preventDefault();
    const formData = {
        template: document.getElementById('template').value || undefined,
        name: document.getElementById('name').value || undefined,
        llm_provider: document.getElementById('provider').value || undefined,
        llm_model: document.getElementById('model').value || undefined,
        plugins: document.getElementById('plugins').value ? document.getElementById('plugins').value.split(',').map(s => s.trim()).filter(Boolean) : undefined,
        communication: document.getElementById('channels').value ? document.getElementById('channels').value.split(',').map(s => s.trim()).filter(Boolean) : undefined,
        languages: document.getElementById('languages').value ? document.getElementById('languages').value.split(',').map(s => s.trim()).filter(Boolean) : ['en'],
        memory: document.getElementById('memory').checked,
        sandbox: document.getElementById('sandbox').checked,
        autostart: document.getElementById('autostart').checked,
    };
    Object.keys(formData).forEach(key => formData[key] === undefined && delete formData[key]);

    try {
        const result = await fetchJSON(`${API_BASE}/agents`, {
            method: 'POST',
            body: JSON.stringify(formData),
        });
        document.getElementById('create-message').innerText = `Created agent ${result.agent_id}`;
        loadAgents();
        loadStatus();
    } catch (err) {
        document.getElementById('create-message').innerText = 'Create failed: ' + err.message;
    }
}

async function startAgent(id) {
    try {
        await fetchJSON(`${API_BASE}/agents/${id}/start`, { method: 'POST' });
        loadAgents();
        loadStatus();
    } catch (err) { alert('Start failed: ' + err.message); }
}

async function stopAgent(id) {
    try {
        await fetchJSON(`${API_BASE}/agents/${id}/stop`, { method: 'POST' });
        loadAgents();
        loadStatus();
    } catch (err) { alert('Stop failed: ' + err.message); }
}

async function deleteAgent(id) {
    if (!confirm('Delete this agent?')) return;
    try {
        await fetchJSON(`${API_BASE}/agents/${id}`, { method: 'DELETE' });
        loadAgents();
        loadStatus();
    } catch (err) { alert('Delete failed: ' + err.message); }
}

async function deployAgent(id) {
    const repoName = prompt('Enter GitHub repo name (optional):');
    try {
        const query = repoName ? `?repo_name=${encodeURIComponent(repoName)}` : '';
        const result = await fetchJSON(`${API_BASE}/agents/${id}/deploy${query}`, { method: 'POST' });
        alert('Deployed! Clone URL: ' + result.clone_url);
    } catch (err) { alert('Deploy failed: ' + err.message); }
}

// ---------- Chat Functions ----------

function addMessage(role, content) {
    const container = document.getElementById('chat-messages');
    const msgDiv = document.createElement('div');
    msgDiv.className = `message ${role}-message`;
    msgDiv.textContent = content;
    container.appendChild(msgDiv);
    container.scrollTop = container.scrollHeight;
    return msgDiv;
}

async function newChat() {
    try {
        const result = await fetchJSON(`${API_BASE}/conversations`, { method: 'POST' });
        currentConversationId = result.conversation_id;
        document.getElementById('chat-messages').innerHTML = '';
        addMessage('assistant', 'New chat started. How can I help you?');
    } catch (err) {
        alert('Failed to create conversation: ' + err.message);
    }
}

async function sendMessage() {
    const input = document.getElementById('chat-input');
    const text = input.value.trim();
    if (!text) return;
    if (!currentConversationId) {
        await newChat();
    }
    addMessage('user', text);
    input.value = '';

    // Prepare request for streaming
    const payload = {
        model: 'gpt-4o-mini',  // default; could be selected elsewhere
        messages: [{ role: 'user', content: text }],
        stream: true,
    };

    // Create streaming assistant message placeholder
    const assistantMsgDiv = addMessage('assistant', '');

    try {
        showActivity();
        const response = await fetch(`${API_BASE}/chat/completions?conversation_id=${currentConversationId}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload),
        });

        if (!response.ok) {
            throw new Error(await response.text());
        }

        const reader = response.body.getReader();
        const decoder = new TextDecoder();
        let buffer = '';

        while (true) {
            const { done, value } = await reader.read();
            if (done) break;
            buffer += decoder.decode(value, { stream: true });
            const lines = buffer.split('\n');
            buffer = lines.pop() || '';
            for (const line of lines) {
                if (line.startsWith('data: ')) {
                    const dataStr = line.slice(6).trim();
                    if (dataStr === '[DONE]') continue;
                    try {
                        const data = JSON.parse(dataStr);
                        const content = data.choices?.[0]?.delta?.content || '';
                        if (content) {
                            assistantMsgDiv.textContent += content;
                            document.getElementById('chat-messages').scrollTop = document.getElementById('chat-messages').scrollHeight;
                        }
                    } catch (e) { /* ignore malformed */ }
                }
            }
        }
    } catch (err) {
        assistantMsgDiv.textContent = 'Error: ' + err.message;
    } finally {
        hideActivity();
    }
}
