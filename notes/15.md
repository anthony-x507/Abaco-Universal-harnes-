# api/schemas.py
"""
Pydantic models for API requests and responses.
"""

from typing import List, Dict, Any, Optional
from pydantic import BaseModel, Field


class ChatMessage(BaseModel):
    role: str
    content: str


class ChatCompletionRequest(BaseModel):
    model: str = "gpt-4o-mini"
    messages: List[ChatMessage]
    temperature: Optional[float] = 0.7
    stream: bool = False
    max_tokens: Optional[int] = None


class ChatCompletionResponse(BaseModel):
    id: str
    object: str = "chat.completion"
    created: int
    model: str
    choices: List[Dict[str, Any]]


class AgentCreateRequest(BaseModel):
    template: Optional[str] = None
    name: Optional[str] = None
    llm_provider: Optional[str] = None
    llm_model: Optional[str] = None
    plugins: Optional[List[str]] = None
    communication: Optional[List[str]] = None
    languages: Optional[List[str]] = None
    system_prompt: Optional[str] = None
    memory: bool = True
    sandbox: bool = False
    autostart: bool = False


class AgentResponse(BaseModel):
    agent_id: str
    name: str
    status: str
    folder: str
    spec: Dict[str, Any]


class PluginExecuteRequest(BaseModel):
    plugin_name: str
    action: str
    parameters: Dict[str, Any] = {}


class CodeExecuteRequest(BaseModel):
    code: str
    language: str = "python"
    timeout: int = 30


class FileUploadResponse(BaseModel):
    file_id: str
    filename: str
    path: str


class FactoryStatusResponse(BaseModel):
    agents: int
    running_agents: List[str]
    plugins: List[str]
    models: List[str]
    communication_channels: List[str]

# api/main.py
"""
FastAPI application for the Aegis Agent Factory.
Exposes endpoints for chat, agent management, plugin execution, and factory status.
"""

import logging
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from .routes import chat, agents, factory

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("api")

# Create FastAPI app
app = FastAPI(
    title="Aegis Agent Factory API",
    description="Backend API for the Aegis Agent Factory, compatible with OpenAI-style UIs.",
    version="0.1.0",
)

# Add CORS middleware to allow local UI connections
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],  # For development; restrict in production
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Include routers
app.include_router(chat.router, prefix="/v1", tags=["chat"])
app.include_router(agents.router, prefix="/v1/agents", tags=["agents"])
app.include_router(factory.router, prefix="/v1", tags=["factory"])


@app.get("/")
async def root():
    return {"message": "Aegis Agent Factory API is running"}


@app.get("/health")
async def health():
    return {"status": "healthy"}


# The actual AgentFactory instance is created in a global variable for simplicity.
# In production, you might use dependency injection or a singleton pattern.

# api/routes/chat.py
"""
OpenAI-compatible chat endpoint.
"""

import time
import uuid
import logging
from fastapi import APIRouter, HTTPException
from ..schemas import ChatCompletionRequest, ChatCompletionResponse
from ..main import factory_instance

router = APIRouter()
logger = logging.getLogger("chat")

@router.post("/chat/completions", response_model=ChatCompletionResponse)
async def chat_completions(request: ChatCompletionRequest):
    """
    Handle chat completion requests.
    This endpoint is compatible with OpenAI's /v1/chat/completions.
    It will forward the messages to a specific agent (or the default main agent)
    and return the response.
    """
    # For simplicity, we use the main agent (or a default created one if not exists).
    # In a full implementation, you would map model names to agent IDs or create a temporary agent.
    # Here we use the factory's main agent if available, else create one.
    # Actually our AgentFactory doesn't have a "main" agent by default; we can create one on first use.
    # For now, we'll just use a temporary Agent instance directly.
    # In a real scenario, you'd likely have a dedicated conversation agent.
    
    # Placeholder: create or use a global agent for chat.
    # We'll implement a simple in-memory agent for demonstration.
    # In production, you would manage conversations and agents properly.
    
    from core.agent import Agent
    import os
    
    # Use environment or config for default agent
    # This is a simplification; not recommended for production.
    if not hasattr(factory_instance, '_chat_agent'):
        config = {
            "agent_id": "chat_agent",
            "agent_name": "Chat Agent",
            "system_prompt": "You are a helpful assistant.",
            "default_llm_provider": request.model.split('/')[0] if '/' in request.model else "openai",
            "default_llm_model": request.model,
            "max_tokens": 8000,
            "token_budget": 10000,
            "data_dir": "data/chat_agent",
        }
        factory_instance._chat_agent = Agent(config=config)
        factory_instance._chat_agent.logger.info("Chat agent created for API")
    
    agent = factory_instance._chat_agent
    
    # Convert messages to the format expected by the agent
    # The agent expects a single user message, but we can also handle conversation history.
    # For simplicity, we take the last user message and ignore history for now.
    # To properly support context, we'd need to integrate with the agent's context manager.
    # For now, we just send the last message.
    
    last_user_message = None
    for msg in reversed(request.messages):
        if msg.role == "user":
            last_user_message = msg.content
            break
    if last_user_message is None:
        raise HTTPException(status_code=400, detail="No user message found.")
    
    # Process the message (non-streaming)
    response_text = agent.process_user_message("api_user", last_user_message)
    
    # Build OpenAI-style response
    response_id = f"chatcmpl-{uuid.uuid4().hex}"
    created = int(time.time())
    response = {
        "id": response_id,
        "object": "chat.completion",
        "created": created,
        "model": request.model,
        "choices": [
            {
                "index": 0,
                "message": {
                    "role": "assistant",
                    "content": response_text,
                },
                "finish_reason": "stop",
            }
        ],
        "usage": {
            "prompt_tokens": 0,
            "completion_tokens": 0,
            "total_tokens": 0,
        },
    }
    return response

# api/routes/agents.py
"""
Agent management endpoints.
"""

import logging
from fastapi import APIRouter, HTTPException
from ..schemas import AgentCreateRequest, AgentResponse, PluginExecuteRequest, CodeExecuteRequest, FileUploadResponse
from ..main import factory_instance
from typing import List

router = APIRouter()
logger = logging.getLogger("agents")

@router.get("", response_model=List[AgentResponse])
async def list_agents():
    """List all agents (running or not)."""
    agents = factory_instance.list_agents()
    return agents

@router.post("", response_model=AgentResponse)
async def create_agent(request: AgentCreateRequest):
    """Create a new agent from the given spec."""
    spec = request.dict(exclude_none=True)
    try:
        agent_info = factory_instance.create_agent(spec)
        return agent_info
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.post("/{agent_id}/start")
async def start_agent(agent_id: str):
    """Start an agent."""
    success = factory_instance.start_agent(agent_id)
    if not success:
        raise HTTPException(status_code=404, detail="Agent not found or already running.")
    return {"status": "started"}

@router.post("/{agent_id}/stop")
async def stop_agent(agent_id: str):
    """Stop an agent."""
    success = factory_instance.stop_agent(agent_id)
    if not success:
        raise HTTPException(status_code=404, detail="Agent not found.")
    return {"status": "stopped"}

@router.get("/{agent_id}/status")
async def get_agent_status(agent_id: str):
    """Get agent status and metrics."""
    status = factory_instance.get_agent_status(agent_id)
    if not status:
        raise HTTPException(status_code=404, detail="Agent not found.")
    # Additional metrics can be added here
    return {"agent_id": agent_id, "status": status}

@router.post("/{agent_id}/deploy")
async def deploy_agent(agent_id: str, repo_name: str = None, private: bool = False):
    """Deploy an agent to GitHub."""
    try:
        clone_url = factory_instance.deploy_to_github(agent_id, repo_name, private)
        return {"clone_url": clone_url}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

# Plugin execution endpoint (global)
@router.post("/plugins/execute")
async def execute_plugin(request: PluginExecuteRequest):
    """Execute an action on a plugin."""
    try:
        result = factory_instance.plugin_manager.execute(request.plugin_name, request.action, request.parameters)
        return {"result": result}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))
