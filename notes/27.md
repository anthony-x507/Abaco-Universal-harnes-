@router.post("/record/ai_create")
async def ai_create_skill(description: str, model: str = "gpt-4o-mini"):
    """
    Use an LLM to generate a sequence of browser actions from a natural language description.
    Returns the actions and optionally saves as a skill.
    """
    # Import LLM provider
    from integrations.llm_providers import get_llm
    provider = get_llm(model.split('/')[0] if '/' in model else "openai", model=model)
    prompt = f"""
    Generate a JSON array of browser actions to accomplish the following task:
    "{description}"
    
    Actions can be:
    - {{"type": "navigate", "url": "https://..."}}
    - {{"type": "click", "selector": "css_selector"}}
    - {{"type": "input", "selector": "css_selector", "value": "text"}}
    - {{"type": "select", "selector": "css_selector", "value": "option_value"}}
    - {{"type": "wait", "seconds": number}}
    
    Return ONLY the JSON array, no additional text.
    """
    messages = [{"role": "user", "content": prompt}]
    response = provider.generate(messages, temperature=0.2)
    # Parse JSON
    import json
    try:
        actions = json.loads(response)
        return {"actions": actions}
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Failed to parse LLM output: {e}")

import json
from pathlib import Path
from plugins.loader import BasePlugin
from sandbox.linux_sandbox import LinuxSandbox  # or direct Playwright if available

class Plugin(BasePlugin):
    def initialize(self, config):
        self.skills_dir = Path(config.get("skills_dir", "data/browser_skills"))
        self.skills_dir.mkdir(parents=True, exist_ok=True)

    def execute(self, action, parameters):
        if action == "execute_skill":
            skill_name = parameters.get("skill_name")
            if not skill_name:
                raise ValueError("skill_name required")
            skill_file = self.skills_dir / f"{skill_name}.json"
            if not skill_file.exists():
                raise FileNotFoundError(f"Skill {skill_name} not found")
            with open(skill_file, 'r') as f:
                actions = json.load(f)
            # In a full integration, this would call the browser API.
            # Here we return the actions to be executed by the agent.
            return {"actions": actions, "status": "ready_to_execute"}
        elif action == "list_skills":
            skills = [f.stem for f in self.skills_dir.glob("*.json")]
            return {"skills": skills}
        else:
            raise NotImplementedError(f"Action '{action}' not supported")

# api/routes/scheduler.py
import asyncio
import json
import time
from pathlib import Path
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from datetime import datetime, timedelta

router = APIRouter()

scheduled_tasks = []

class ScheduleRequest(BaseModel):
    skill_name: str
    run_at: str  # ISO datetime or "now"
    repeat_interval_minutes: int = 0  # 0 = run once

@router.post("/scheduler/add")
async def add_schedule(request: ScheduleRequest):
    task = {
        "skill_name": request.skill_name,
        "run_at": request.run_at,
        "repeat_interval_minutes": request.repeat_interval_minutes,
        "status": "scheduled",
    }
    scheduled_tasks.append(task)
    # Start background task if not already running
    asyncio.create_task(_scheduler_loop())
    return {"status": "scheduled", "task": task}

async def _scheduler_loop():
    while True:
        now = datetime.utcnow()
        for task in scheduled_tasks:
            if task["status"] != "scheduled":
                continue
            run_at = datetime.fromisoformat(task["run_at"])
            if now >= run_at:
                # Execute skill
                try:
                    # Call the browser skill execution (simplified: just log)
                    # In a real implementation, call the browser API or plugin.
                    task["status"] = "executed"
                    task["last_run"] = now.isoformat()
                except Exception as e:
                    task["status"] = "error"
                    task["error"] = str(e)
        await asyncio.sleep(30)
