# integrations/llm_providers.py
"""
Unified LLM Provider Manager.
Supports 25+ providers through a common interface.
Each provider can be selected by name when creating an agent.
"""

import os
import logging
from typing import Dict, Any, List, Optional


class BaseLLMProvider:
    """Base class for all LLM providers."""

    api_key_env: str = ""
    default_model: str = ""

    def __init__(self, api_key: Optional[str] = None, model: Optional[str] = None):
        self.api_key = api_key or os.getenv(self.api_key_env, "")
        self.model = model or self.default_model
        self.logger = logging.getLogger(self.__class__.__name__)

    def generate(self, messages: List[Dict[str, str]], **kwargs) -> str:
        raise NotImplementedError

    def stream(self, messages: List[Dict[str, str]], **kwargs):
        raise NotImplementedError


class OpenAIProvider(BaseLLMProvider):
    api_key_env = "OPENAI_API_KEY"
    default_model = "gpt-4o-mini"

    def generate(self, messages, **kwargs):
        import openai
        openai.api_key = self.api_key
        response = openai.ChatCompletion.create(
            model=self.model,
            messages=messages,
            **kwargs
        )
        return response.choices[0].message.content.strip()


class OpenAICompatibleProvider(BaseLLMProvider):
    """Generic provider for services with OpenAI-compatible API."""
    base_url: str = ""
    api_key_env: str = ""
    default_model: str = ""

    def generate(self, messages, **kwargs):
        import openai
        openai.api_key = self.api_key
        openai.api_base = self.base_url
        response = openai.ChatCompletion.create(
            model=self.model,
            messages=messages,
            **kwargs
        )
        return response.choices[0].message.content.strip()


class AnthropicProvider(BaseLLMProvider):
    api_key_env = "ANTHROPIC_API_KEY"
    default_model = "claude-3-5-sonnet-20241022"

    def generate(self, messages, **kwargs):
        import anthropic
        client = anthropic.Anthropic(api_key=self.api_key)
        system = ""
        if messages and messages[0]["role"] == "system":
            system = messages[0]["content"]
            messages = messages[1:]
        response = client.messages.create(
            model=self.model,
            system=system,
            messages=messages,
            **kwargs
        )
        return response.content[0].text.strip()


class GoogleProvider(BaseLLMProvider):
    api_key_env = "GOOGLE_API_KEY"
    default_model = "gemini-1.5-pro"

    def generate(self, messages, **kwargs):
        import google.generativeai as genai
        genai.configure(api_key=self.api_key)
        model = genai.GenerativeModel(self.model)
        prompt = "\n".join([f"{m['role']}: {m['content']}" for m in messages])
        response = model.generate_content(prompt)
        return response.text.strip()

class HuggingFaceProvider(BaseLLMProvider):
    api_key_env = "HF_API_KEY"
    default_model = "meta-llama/Llama-3.1-8B-Instruct"

    def generate(self, messages, **kwargs):
        import requests
        prompt = "\n".join([f"{m['role']}: {m['content']}" for m in messages])
        headers = {"Authorization": f"Bearer {self.api_key}"}
        payload = {"inputs": prompt, "parameters": kwargs}
        response = requests.post(
            f"https://api-inference.huggingface.co/models/{self.model}",
            headers=headers, json=payload, timeout=60
        )
        data = response.json()
        if isinstance(data, list) and data:
            return data[0].get("generated_text", "").replace(prompt, "").strip()
        return str(data)

    provider_class = PROVIDER_REGISTRY.get(provider_name.lower())

"""
Additional communication channel implementations.
Each class inherits from BaseCommunication and provides send_message.
Many are skeletons requiring specific libraries.

import logging
from .base import BaseCommunication


class WhatsAppTwilioChannel(BaseCommunication):
    def send_message(self, recipient, text):
        try:
            from twilio.rest import Client
            account_sid = self.config.get("account_sid") or __import__('os').environ.get("TWILIO_ACCOUNT_SID")
            auth_token = self.config.get("auth_token") or __import__('os').environ.get("TWILIO_AUTH_TOKEN")
            client = Client(account_sid, auth_token)
            message = client.messages.create(
                body=text,
                from_=self.config.get("from_whatsapp_number"),
                to=f"whatsapp:{recipient}"
            )
            return True
        except Exception as e:
            self.logger.error(f"WhatsApp send failed: {e}")
            return False


class SignalChannel(BaseCommunication):
    def send_message(self, recipient, text):
        # Uses signal-cli or pysignal
        try:
            import subprocess
            cmd = ["signal-cli", "-u", self.config.get("username"), "send", "-m", text, recipient]
            subprocess.run(cmd, check=True)
            return True
        except Exception as e:
            self.logger.error(f"Signal send failed: {e}")
            return False


class MatrixChannel(BaseCommunication):
    def send_message(self, recipient, text):
        try:
            from matrix_client.client import MatrixClient
            client = MatrixClient(self.config.get("homeserver"))
            token = client.login_with_password(self.config.get("username"), self.config.get("password"))
            room = client.join_room(recipient)
            room.send_text(text)
            return True
        except Exception as e:
            self.logger.error(f"Matrix send failed: {e}")
            return False


class IRCChannel(BaseCommunication):
    def send_message(self, recipient, text):
        # recipient is channel like "#room"
        try:
            import socket
            server = self.config.get("server")
            port = self.config.get("port", 6667)
            nick = self.config.get("nick", "aegis")
            s = socket.socket()
            s.connect((server, port))
            s.send(f"NICK {nick}\r\n".encode())
            s.send(f"USER {nick} 0 * :{nick}\r\n".encode())
            s.send(f"JOIN {recipient}\r\n".encode())
            s.send(f"PRIVMSG {recipient} :{text}\r\n".encode())
            s.send(b"QUIT\r\n")
            s.close()
            return True
        except Exception as e:
            self.logger.error(f"IRC send failed: {e}")
            return False
