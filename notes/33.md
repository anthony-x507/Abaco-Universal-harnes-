1. Backend: Natural Language Scheduling Endpoint

We'll add an endpoint in api/routes/scheduler.py that accepts a description, uses the LLM to extract scheduling details, and then adds the task.

Update api/routes/scheduler.py

# api/routes/scheduler.py
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import Optional
from tools.browser_scheduler import BrowserScheduler
from integrations.llm_providers import get_llm

router = APIRouter()
scheduler = BrowserScheduler()

class ScheduleRequest(BaseModel):
    skill_name: str
    run_at: Optional[str] = None
    cron: Optional[str] = None
    timezone: str = "UTC"
    repeat_interval_minutes: int = 0

class NLScheduleRequest(BaseModel):
    description: str
    model: str = "gpt-4o-mini"   # provider/model for parsing

@router.post("/scheduler/nl_add")
async def nl_add_schedule(request: NLScheduleRequest):
    """
    Use an LLM to parse a natural language scheduling instruction and add the task.
    Example: "Run the login_to_dashboard skill every day at 9am UTC"
    """
    # 1. Get LLM provider
    provider = get_llm(request.model.split('/')[0] if '/' in request.model else "openai",
                       model=request.model)
    # 2. Build prompt to extract structured info
    prompt = f"""
    Parse the following scheduling instruction and return a JSON object with these fields:
    - skill_name: the name of the browser skill to run (string)
    - cron: cron expression (string) if the instruction specifies a recurring schedule, otherwise null
    - run_at: ISO datetime (string) if it's a one-time schedule, otherwise null
    - timezone: timezone (string), default "UTC" if not specified

    Instruction: "{request.description}"

    Return ONLY the JSON object, no additional text.
    """
    messages = [{"role": "user", "content": prompt}]
    response = provider.generate(messages, temperature=0.1)
    # 3. Parse JSON
    import json
    try:
        parsed = json.loads(response)
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Failed to parse LLM output: {e}")

    # 4. Validate required field
    skill_name = parsed.get("skill_name")
    if not skill_name:
        raise HTTPException(status_code=400, detail="skill_name not found in parsing result")

    # 5. Add task using existing scheduler
    try:
        task = scheduler.add_task(
            skill_name=skill_name,
            run_at=parsed.get("run_at"),
            cron=parsed.get("cron"),
            timezone=parsed.get("timezone", "UTC"),
            repeat_interval_minutes=0,  # cron/run_at handles recurrence
        )
        return {"status": "scheduled", "task": task, "parsed": parsed}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

Note: We assume the browser skill names are known and exist; if not, the scheduler will fail at execution. We could add a check, but for now it's fine.
