# tests/test_integration.py
"""
End-to-end integration tests for the Aegis Agent Factory API.

Run with: pytest tests/test_integration.py -v
"""

import sys
import os
import json
import pytest
from pathlib import Path
from fastapi.testclient import TestClient

# Ensure project root is in sys.path
sys.path.insert(0, str(Path(__file__).parent.parent))

from api.main import app, factory_instance
from api.conversation_store import conversation_store

client = TestClient(app)

@pytest.fixture(autouse=True)
def cleanup():
    """Reset factory and conversation store before each test."""
    # Stop all running agents
    for agent_id in factory_instance.list_running_agents():
        factory_instance.stop_agent(agent_id)
    # Clear conversation store
    conversation_store.conversations.clear()
    # Clean up data directories (optional, but keep for simplicity)
    yield
    # Additional cleanup after test
    for agent_id in list(factory_instance.registry._read().keys()):
        factory_instance.delete_agent(agent_id)


def test_health():
    response = client.get("/health")
    assert response.status_code == 200
    assert response.json()["status"] == "healthy"

def test_models_list():
    response = client.get("/v1/models")
    assert response.status_code == 200
    data = response.json()
    assert "providers" in data
    assert len(data["providers"]) >= 1

def test_create_agent():
    agent_spec = {
        "name": "Test Agent",
        "llm_provider": "openai",
        "llm_model": "gpt-4o-mini",
        "autostart": False,
    }
    response = client.post("/v1/agents", json=agent_spec)
    assert response.status_code == 200, response.text
    data = response.json()
    assert data["agent_id"]
    assert data["name"] == "Test Agent"
    # Verify agent appears in list
    list_resp = client.get("/v1/agents")
    assert any(a["agent_id"] == data["agent_id"] for a in list_resp.json())

def test_start_agent():
    # Create agent
    agent_spec = {"name": "Start Test", "llm_provider": "openai"}
    create_resp = client.post("/v1/agents", json=agent_spec)
    agent_id = create_resp.json()["agent_id"]
    # Start agent
    start_resp = client.post(f"/v1/agents/{agent_id}/start")
    assert start_resp.status_code == 200
    # Check status
    status_resp = client.get(f"/v1/agents/{agent_id}/status")
    assert status_resp.status_code == 200
    assert status_resp.json()["status"] == "running"

def test_chat_completion_non_streaming():
    # Use the default chat agent, no need to create
    response = client.post(
        "/v1/chat/completions",
        json={
            "model": "gpt-4o-mini",
            "messages": [{"role": "user", "content": "Hello"}],
            "stream": False,
        },
    )
    assert response.status_code == 200
    data = response.json()
    assert "choices" in data
    assert len(data["choices"]) > 0
    assert data["choices"][0]["message"]["content"]

def test_plugin_list():
    response = client.get("/v1/plugins")
    assert response.status_code == 200
    data = response.json()
    assert "plugins" in data
    # Plugins may be empty if not loaded; that's fine, but ensure list type
    assert isinstance(data["plugins"], list)

def test_scheduler_add_and_list():
    # Add a one-time task that will never run (run_at in far future)
    run_at = "2099-01-01T00:00:00"
    response = client.post(
        "/v1/scheduler/add",
        json={"skill_name": "test_skill", "run_at": run_at},
    )
    assert response.status_code == 200
    task = response.json()["task"]
    assert task["skill_name"] == "test_skill"
    # List tasks
    list_resp = client.get("/v1/scheduler/list")
    assert any(t["skill_name"] == "test_skill" for t in list_resp.json()["tasks"])

def test_browser_state_endpoints():
    # Test skill state save/load without launching browser
    skill_name = "test_skill_state"
    state = {"last_page": "dashboard", "counter": 5}
    # Save state
    put_resp = client.put(f"/v1/browser/skills/{skill_name}/state", json=state)
    assert put_resp.status_code == 200
    # Load state
    get_resp = client.get(f"/v1/browser/skills/{skill_name}/state")
    assert get_resp.status_code == 200
    data = get_resp.json()
    assert data["exists"] is True
    assert data["state"] == state

@pytest.mark.skipif(not os.environ.get("PLAYWRIGHT_INSTALLED"), reason="Playwright not installed")
def test_browser_launch_and_navigate():
    # This test is optional; requires Playwright and browser binaries.
    launch_resp = client.post("/v1/browser/launch")
    assert launch_resp.status_code == 200
    nav_resp = client.post("/v1/browser/navigate", json={"url": "https://example.com"})
    assert nav_resp.status_code == 200
    screenshot_resp = client.get("/v1/browser/screenshot")
    assert screenshot_resp.status_code == 200
    assert "image" in screenshot_resp.json()

5. playwright install chromium
