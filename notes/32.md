1. Backend: State Endpoints in api/routes/browser.py

Add two endpoints to get and update skill state.

@router.get("/skills/{skill_name}/state")
async def get_skill_state(skill_name: str):
    state_file = Path("data/browser_state") / f"{skill_name}_state.json"
    if not state_file.exists():
        return {"state": {}, "exists": False}
    with open(state_file, 'r') as f:
        state = json.load(f)
    return {"state": state, "exists": True}

@router.put("/skills/{skill_name}/state")
async def update_skill_state(skill_name: str, state: Dict[str, Any]):
    state_file = Path("data/browser_state") / f"{skill_name}_state.json"
    state_file.parent.mkdir(parents=True, exist_ok=True)
    with open(state_file, 'w') as f:
        json.dump(state, f, indent=2)
    return {"status": "updated", "state": state}

These endpoints allow both UI and agents to read/write state via HTTP.

# plugins/browser_scheduler_plugin/plugin.py (add actions)
import requests
import json
from pathlib import Path

class Plugin(BasePlugin):
    # existing methods...
    def execute(self, action, parameters):
        if action == "schedule_skill":
            # ... existing
        elif action == "list_scheduled":
            # ... existing
        elif action == "delete_scheduled":
            # ... existing
        elif action == "get_skill_state":
            skill_name = parameters.get("skill_name")
            if not skill_name:
                raise ValueError("skill_name required")
            response = requests.get(f"{self.scheduler.api_base_url}/v1/browser/skills/{skill_name}/state")
            if response.status_code == 200:
                return response.json()
            else:
                return {"error": f"Failed to get state: {response.status_code}"}
        elif action == "update_skill_state":
            skill_name = parameters.get("skill_name")
            state = parameters.get("state")
            if not skill_name or state is None:
                raise ValueError("skill_name and state required")
            response = requests.put(
                f"{self.scheduler.api_base_url}/v1/browser/skills/{skill_name}/state",
                json=state
            )
            if response.status_code == 200:
                return response.json()
            else:
                return {"error": f"Failed to update state: {response.status_code}"}
        else:
            raise NotImplementedError(f"Action '{action}' not supported")

async function loadSkillsForState() {
    try {
        const data = await fetchJSON(`${API_BASE}/browser/record/list`);
        const select = document.getElementById('state-skill-select');
        select.innerHTML = '<option value="">Select skill...</option>';
        data.skills.forEach(skill => {
            const opt = document.createElement('option');
            opt.value = skill;
            opt.textContent = skill;
            select.appendChild(opt);
        });
    } catch (e) { console.error('Failed to load skills for state', e); }
}

async function loadStateForSkill() {
    const skill = document.getElementById('state-skill-select').value;
    if (!skill) return;
    try {
        const data = await fetchJSON(`${API_BASE}/browser/skills/${skill}/state`);
        document.getElementById('state-editor').value = JSON.stringify(data.state, null, 2);
    } catch (e) { alert('Failed to load state: ' + e.message); }
}

async function saveStateForSkill() {
    const skill = document.getElementById('state-skill-select').value;
    if (!skill) return;
    let state;
    try {
        state = JSON.parse(document.getElementById('state-editor').value);
    } catch (e) {
        alert('Invalid JSON');
        return;
    }
    try {
        await fetchJSON(`${API_BASE}/browser/skills/${skill}/state`, {
            method: 'PUT',
            body: JSON.stringify(state),
        });
        alert('State saved');
    } catch (e) { alert('Save failed: ' + e.message); }
}

document.getElementById('state-load-btn').addEventListener('click', loadStateForSkill);
document.getElementById('state-save-btn').addEventListener('click', saveStateForSkill);
