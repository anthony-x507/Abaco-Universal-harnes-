1. Update Browser Replay Endpoint to Handle State

Modify api/routes/browser.py → record_replay

We'll change the endpoint to accept an optional state (JSON) and return updated state after execution. The browser actions themselves may not update state, but we can allow the skill to include special actions to modify state (e.g., set_state, get_state).

@router.post("/record/replay")
async def record_replay(skill_name: str, state: Optional[Dict[str, Any]] = None):
    page = get_active_page()
    if not page:
        raise HTTPException(status_code=400, detail="No active tab")
    skills_dir = Path("data/browser_skills")
    file_path = skills_dir / f"{skill_name}.json"
    if not file_path.exists():
        raise HTTPException(status_code=404, detail="Skill not found")
    with open(file_path, 'r') as f:
        actions = json.load(f)
    # Load state from file if not provided
    if state is None:
        state_file = Path("data/browser_state") / f"{skill_name}_state.json"
        if state_file.exists():
            with open(state_file, 'r') as f:
                state = json.load(f)
        else:
            state = {}
    results = []
    for action in actions:
        try:
            if action['type'] == 'navigate':
                await page.goto(action['url'])
            elif action['type'] == 'click':
                await page.click(action['selector'], timeout=2000)
            elif action['type'] == 'input':
                await page.fill(action['selector'], action.get('value', ''))
            elif action['type'] == 'select':
                await page.select_option(action['selector'], action.get('value', ''))
            elif action['type'] == 'wait':
                await asyncio.sleep(action.get('seconds', 1))
            elif action['type'] == 'set_state':
                # Allow skill to set state variables
                key = action.get('key')
                value = action.get('value')
                if key:
                    state[key] = value
            results.append({"status": "ok", "action": action})
        except Exception as e:
            results.append({"status": "error", "action": action, "error": str(e)})
    # Save updated state
    state_file = Path("data/browser_state") / f"{skill_name}_state.json"
    state_file.parent.mkdir(parents=True, exist_ok=True)
    with open(state_file, 'w') as f:
        json.dump(state, f, indent=2)
    return {"results": results, "state": state}

In _execute_task, before calling replay, we load the state file for that skill and include it in the request. After the response, we save the returned state.

def _execute_task(self, task: Dict, now: datetime):

    state_file = Path(f"data/browser_state/{skill_name}_state.json")
    state = {}
    if state_file.exists():
        try:
            with open(state_file, 'r') as f:
                state = json.load(f)
        except:
            pass

    try:

            params={"skill_name": skill_name},
            json=state,  # pass state as JSON body
            timeout=60,
        )

            task["status"] = "executed"
            task["last_run"] = now.isoformat()

            task["error"] = f"Replay failed with status {response.status_code}"

✅ Summary of Part 33

We have implemented state persistence for scheduled browser skills:

· The replay endpoint loads and saves a JSON state file per skill.
