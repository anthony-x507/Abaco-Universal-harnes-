1. Create tools/browser_scheduler.py

This module handles task storage, background execution via internal HTTP calls, and provides methods for agents.

# tools/browser_scheduler.py
"""
Shared browser skill scheduler.
Used by API routes and agent plugins to schedule and manage browser skill executions.
"""

import threading
import time
import requests
import json
from datetime import datetime, timedelta
from typing import List, Dict, Any, Optional

class BrowserScheduler:
    def __init__(self, api_base_url="http://localhost:8000"):
        self.api_base_url = api_base_url
        self.tasks: List[Dict[str, Any]] = []
        self.lock = threading.Lock()
        self.thread = threading.Thread(target=self._run_loop, daemon=True)
        self.thread.start()

    def add_task(self, skill_name: str, run_at: str, repeat_interval_minutes: int = 0) -> Dict:
        task = {
            "skill_name": skill_name,
            "run_at": run_at,
            "repeat_interval_minutes": repeat_interval_minutes,
            "status": "scheduled",
            "last_run": None,
            "error": None,
        }
        with self.lock:
            self.tasks.append(task)
        return task

    def get_tasks(self) -> List[Dict]:
        with self.lock:
            return list(self.tasks)

    def delete_task(self, index: int) -> Optional[Dict]:
        with self.lock:
            if 0 <= index < len(self.tasks):
                return self.tasks.pop(index)
        return None

    def _run_loop(self):
        while True:
            now = datetime.utcnow()
            with self.lock:
                for task in self.tasks:
                    if task["status"] not in ("scheduled", "retry"):
                        continue
                    run_at = datetime.fromisoformat(task["run_at"])
                    if now >= run_at:
                        self._execute_task(task, now)
            time.sleep(10)  # check every 10 seconds

    def _execute_task(self, task: Dict, now: datetime):
        """Execute a task by calling the browser replay API endpoint."""
        try:
            # Use requests to call our own API (thread-safe)
            response = requests.post(
                f"{self.api_base_url}/v1/browser/record/replay",
                params={"skill_name": task["skill_name"]},
                timeout=60,
            )
            if response.status_code == 200:
                task["status"] = "executed"
                task["last_run"] = now.isoformat()
                task["error"] = None
            else:
                task["status"] = "error"
                task["error"] = f"Replay failed with status {response.status_code}"
        except Exception as e:
            task["status"] = "error"
            task["error"] = str(e)

        # Handle repeat
        if task["repeat_interval_minutes"] > 0:
            next_run = now + timedelta(minutes=task["repeat_interval_minutes"])
            task["run_at"] = next_run.isoformat()
            task["status"] = "scheduled"

2. Update api/routes/scheduler.py to use the shared scheduler

We'll replace the existing route implementation with one that delegates to BrowserScheduler.

# api/routes/scheduler.py
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from tools.browser_scheduler import BrowserScheduler

router = APIRouter()

# Singleton scheduler instance (same as used by plugins)
scheduler = BrowserScheduler()

class ScheduleRequest(BaseModel):
    skill_name: str
    run_at: str
    repeat_interval_minutes: int = 0

@router.post("/scheduler/add")
async def add_schedule(request: ScheduleRequest):
    task = scheduler.add_task(request.skill_name, request.run_at, request.repeat_interval_minutes)
    return {"status": "scheduled", "task": task}

@router.get("/scheduler/list")
async def list_schedules():
    return {"tasks": scheduler.get_tasks()}

@router.delete("/scheduler/{task_index}")
async def delete_schedule(task_index: int):
    removed = scheduler.delete_task(task_index)
    if removed is None:
        raise HTTPException(status_code=404, detail="Task not found")
    return {"removed": removed}

# plugins/browser_scheduler_plugin/plugin.py
import json
from datetime import datetime
from plugins.loader import BasePlugin
from tools.browser_scheduler import BrowserScheduler

class Plugin(BasePlugin):
    """Allows agents to schedule browser skills."""

    def initialize(self, config):
        # Use the same scheduler instance as the API
        self.scheduler = BrowserScheduler()
        self.scheduler.api_base_url = config.get("api_base_url", "http://localhost:8000")

    def execute(self, action, parameters):
        if action == "schedule_skill":
            skill_name = parameters.get("skill_name")
            run_at = parameters.get("run_at")
            repeat_interval_minutes = parameters.get("repeat_interval_minutes", 0)
            if not skill_name or not run_at:
                raise ValueError("skill_name and run_at required")
            task = self.scheduler.add_task(skill_name, run_at, repeat_interval_minutes)
            return {"status": "scheduled", "task": task}
        elif action == "list_scheduled":
            return {"tasks": self.scheduler.get_tasks()}
        elif action == "delete_scheduled":
            index = parameters.get("index")
            if index is None:
                raise ValueError("index required")
            removed = self.scheduler.delete_task(int(index))
            return {"removed": removed}
        else:
            raise NotImplementedError(f"Action '{action}' not supported")

    "run_at": "2025-03-20T09:00:00",

This will add the task to the shared scheduler, and the background thread will execute it at the specified time.
