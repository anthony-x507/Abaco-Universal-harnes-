# plugins/ew_jammers/plugin.py
import subprocess
import json
import logging
from pathlib import Path
from plugins.loader import BasePlugin

class Plugin(BasePlugin):
    """Electronic Warfare / Jammers plugin."""

    def initialize(self, config):
        self.config = config
        self.sdr_device = config.get("sdr_device", "hackrf")
        self.default_power = config.get("default_power", 10)
        self.log_dir = Path(config.get("log_dir", "data/ew_logs"))
        self.log_dir.mkdir(parents=True, exist_ok=True)
        self.logger = logging.getLogger("ew_jammers")

    def execute(self, action, parameters):
        if action == "start_jamming":
            return self._start_jamming(parameters)
        elif action == "stop_jamming":
            return self._stop_jamming(parameters)
        elif action == "detect_jamming":
            return self._detect_jamming(parameters)
        elif action == "electronic_protection":
            return self._electronic_protection(parameters)
        elif action == "counter_drone":
            return self._counter_drone(parameters)
        else:
            raise NotImplementedError(f"Action '{action}' not supported by ew_jammers")

    def _check_authorization(self):
        """Warn user and require confirmation before any jamming action."""
        # In real UI, this would prompt the user. For now, log a warning.
        self.logger.warning("Jamming is illegal in most jurisdictions without explicit authorization.")
        return True  # placeholder; actual confirmation handled by UI

    def _start_jamming(self, params):
        """
        Start jamming a specific frequency band using HackRF or similar.
        params: frequency, bandwidth, power, duration
        """
        if not self._check_authorization():
            return {"status": "cancelled"}
        freq = params.get("frequency", 2400000000)  # 2.4 GHz
        bandwidth = params.get("bandwidth", 20000000)  # 20 MHz
        power = params.get("power", self.default_power)
        duration = params.get("duration", 10)  # seconds

        # Use hackrf_transfer for transmission
        cmd = [
            "hackrf_transfer",
            "-f", str(freq),
            "-s", str(bandwidth),
            "-a", "1",  # enable amp
            "-x", str(power),
            "-t", str(duration)
        ]
        proc = subprocess.Popen(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return {
            "status": "jamming_started",
            "pid": proc.pid,
            "frequency": freq,
            "bandwidth": bandwidth,
            "duration": duration
        }

    def _stop_jamming(self, params):
        """Stop any running jamming process."""
        subprocess.run(["pkill", "hackrf_transfer"], capture_output=True)
        return {"status": "jamming_stopped"}

# Electronic Warfare & Jammers Module

This module provides agents with tools for signal jamming (authorized use only), jamming detection, electronic protection, and counter-drone measures. It integrates with SDR for both transmission and reception.

## Supported Actions
- `start_jamming` – begin jamming on a specified frequency (requires HackRF/BladeRF).
- `stop_jamming` – stop all jamming processes.
- `detect_jamming` – detect potential jamming by spectrum analysis (placeholder).
- `electronic_protection` – provide recommendations for protecting against jamming/spoofing.
- `counter_drone` – prepare non-kinetic counter-drone measures (jamming control link).

## Requirements
- **Hardware**: HackRF One (or BladeRF) for transmission; RTL-SDR for detection.
- **Software**: `hackrf`, `rtl-sdr`, `pyhackrf` (installed in sandbox or host).
- **Permissions**: USB access for SDR devices.

## Usage Example
1. Agent loads `ew_specialist` template.
2. User (authorized) asks: "Test my GPS receiver against jamming."
3. Agent calls `start_jamming` with frequency=1575420000 (GPS L1) and duration=30.
4. User observes GPS loss; agent then calls `electronic_protection` for recommendations.
5. Agent logs the event and suggests mitigation.

## Legal & Safety Warning
- **Jamming is illegal in most countries without explicit government authorization.**
- Use this module only in controlled, authorized environments (e.g., shielded lab, military training).
- The agent must always verify authorization before executing any jamming action.
- Electronic protection measures should be implemented to defend against illegal jamming.

## Extending
- Add GPS spoofing generation (for receiver testing).
- Integrate direction-finding (TDOA/AOA) for locating jammers.
- Implement automated jamming detection using machine learning on spectrum data.
