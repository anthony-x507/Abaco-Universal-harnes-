# factory/registry.py
"""
Agent Registry - tracks all created agents in a JSON file.
"""

import json
import os
from pathlib import Path
from typing import Dict, Any, List, Optional
import datetime


class AgentRegistry:
    """Simple file-based registry for created agents."""

    def __init__(self, registry_file: str = "data/factory_registry.json"):
        self.registry_file = Path(registry_file)
        self.registry_file.parent.mkdir(parents=True, exist_ok=True)
        self._ensure_file()

    def _ensure_file(self):
        """Create registry file if it doesn't exist."""
        if not self.registry_file.exists():
            self._write({})

    def _read(self) -> Dict[str, Any]:
        """Read registry data from file."""
        try:
            with open(self.registry_file, "r", encoding="utf-8") as f:
                return json.load(f)
        except (FileNotFoundError, json.JSONDecodeError):
            return {}

    def _write(self, data: Dict[str, Any]):
        """Write registry data to file."""
        with open(self.registry_file, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

    def add_agent(self, agent_id: str, agent_info: Dict[str, Any]) -> None:
        """Add or update an agent entry."""
        data = self._read()
        data[agent_id] = agent_info
        data[agent_id]["updated_at"] = datetime.datetime.now().isoformat()
        self._write(data)

    def get_agent(self, agent_id: str) -> Optional[Dict[str, Any]]:
        """Return agent info if exists."""
        data = self._read()
        return data.get(agent_id)

    def list_agents(self) -> List[Dict[str, Any]]:
        """Return list of all agent info dicts."""
        data = self._read()
        return list(data.values())

    def remove_agent(self, agent_id: str) -> bool:
        """Remove an agent from registry."""
        data = self._read()
        if agent_id in data:
            del data[agent_id]
            self._write(data)
            return True
        return False

    def update_status(self, agent_id: str, status: str) -> None:
        """Update the status of an agent."""
        data = self._read()
        if agent_id in data:
            data[agent_id]["status"] = status
            data[agent_id]["updated_at"] = datetime.datetime.now().isoformat()
            self._write(data)

# factory/lifecycle.py
"""
Lifecycle Manager - handles starting, stopping, pausing agents.
Agents run in the same process for now; we keep references in memory.
"""

import logging
from typing import Dict, Any, Optional
from core.agent import Agent


class LifecycleManager:
    """Manages agent instances (in-memory)."""

    def __init__(self):
        self.agents: Dict[str, Agent] = {}
        self.logger = logging.getLogger("lifecycle")

    def start_agent(self, agent_id: str, config: Dict[str, Any]) -> Agent:
        """
        Create and start an agent instance.
        If already running, return existing.
        """
        if agent_id in self.agents:
            return self.agents[agent_id]
        # Create new Agent instance using the provided config
        agent = Agent(config=config)
        self.agents[agent_id] = agent
        self.logger.info(f"Agent {agent_id} started.")
        return agent

    def stop_agent(self, agent_id: str) -> bool:
        """Stop and remove an agent from memory."""
        if agent_id in self.agents:
            del self.agents[agent_id]
            self.logger.info(f"Agent {agent_id} stopped.")
            return True
        return False

    def get_agent(self, agent_id: str) -> Optional[Agent]:
        """Return a running agent instance."""
        return self.agents.get(agent_id)

    def list_running_agents(self) -> list:
        """Return list of running agent IDs."""
        return list(self.agents.keys())

    def pause_agent(self, agent_id: str) -> bool:
        """Placeholder for future pause functionality."""
        # In-memory agents cannot be truly paused; we just log.
        self.logger.info(f"Pause requested for {agent_id} (not implemented).")
        return False

# factory/generator.py
"""
Agent Generator - creates agent instances from specifications.
Handles template loading, validation, and folder creation.
"""

import os
import yaml
import uuid
import logging
from pathlib import Path
from typing import Dict, Any, Optional, List
from core.agent import Agent
from .registry import AgentRegistry
from .lifecycle import LifecycleManager


class AgentGenerator:
    """Generates agents from specs (dict) or templates."""

    def __init__(self, base_data_dir: str = "data/agents", templates_dir: str = "factory/templates"):
        self.base_data_dir = Path(base_data_dir)
        self.base_data_dir.mkdir(parents=True, exist_ok=True)
        self.templates_dir = Path(templates_dir)
        self.logger = logging.getLogger("generator")
        self.registry = AgentRegistry()
        self.lifecycle = LifecycleManager()

    def load_template(self, template_name: str) -> Dict[str, Any]:
        """Load a YAML template file."""
        template_file = self.templates_dir / f"{template_name}.yaml"
        if not template_file.exists():
            raise FileNotFoundError(f"Template '{template_name}' not found.")
        with open(template_file, "r", encoding="utf-8") as f:
            template = yaml.safe_load(f)
        return template or {}

    def create_agent(self, spec: Dict[str, Any]) -> Dict[str, Any]:
        """
        Create a new agent from a specification dict.
        spec may include 'template' key to load a template first,
        then override with any other keys.
        Returns agent info dict (including id, folder, etc.)
        """
        # If template specified, load and merge
        if "template" in spec:
            template_name = spec.pop("template")
            template_spec = self.load_template(template_name)
            # Merge: template first, then user spec overrides
            merged_spec = {**template_spec, **spec}
        else:
            merged_spec = spec

        # Generate unique agent id
        agent_id = merged_spec.get("agent_id") or str(uuid.uuid4())
        # Ensure we don't overwrite existing agent if id given
        if self.registry.get_agent(agent_id):
            raise ValueError(f"Agent with id '{agent_id}' already exists.")

        # Create agent folder
        agent_folder = self.base_data_dir / agent_id
        agent_folder.mkdir(parents=True, exist_ok=True)

        # Write config.yaml (the agent-specific config)
        config_file = agent_folder / "config.yaml"
        with open(config_file, "w", encoding="utf-8") as f:
            yaml.dump(merged_spec, f)

        # Write system_prompt.txt if present
        if "system_prompt" in merged_spec:
            prompt_file = agent_folder / "system_prompt.txt"
            with open(prompt_file, "w", encoding="utf-8") as f:
                f.write(merged_spec["system_prompt"])

        # Register the agent in registry
        agent_info = {
            "agent_id": agent_id,
            "name": merged_spec.get("name", f"Agent_{agent_id[:6]}"),
            "created_at": __import__("datetime").datetime.now().isoformat(),
            "status": "created",
            "folder": str(agent_folder),
            "spec": merged_spec,
        }
        self.registry.add_agent(agent_id, agent_info)
        self.logger.info(f"Agent {agent_id} generated at {agent_folder}")

        # Optionally start the agent immediately
        if merged_spec.get("autostart", False):
            self.lifecycle.start_agent(agent_id, merged_spec)
            self.registry.update_status(agent_id, "running")
            agent_info["status"] = "running"

        return agent_info

    def delete_agent(self, agent_id: str) -> bool:
        """Delete an agent: stop it and remove folder."""
        # Stop if running
        self.lifecycle.stop_agent(agent_id)
        # Remove registry entry
        self.registry.remove_agent(agent_id)
        # Remove folder
        agent_folder = self.base_data_dir / agent_id
        if agent_folder.exists():
            import shutil
            shutil.rmtree(agent_folder)
            self.logger.info(f"Agent {agent_id} deleted.")
            return True
        return False

# factory/manager.py
"""
Agent Factory Manager - central facade for creating and managing agents.
"""

import logging
from typing import Dict, Any, Optional, List
from .generator import AgentGenerator
from .registry import AgentRegistry
from .lifecycle import LifecycleManager
from .packager import PackageBuilder


class AgentFactory:
    """
    Main entry point for the Agent Factory.
    Provides high-level methods for agent creation, lifecycle, and deployment.
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None):
        self.config = config or {}
        self.logger = logging.getLogger("factory")
        self.registry = AgentRegistry()
        self.lifecycle = LifecycleManager()
        self.generator = AgentGenerator()
        self.packager = PackageBuilder()

    def create_agent(self, spec: Dict[str, Any]) -> Dict[str, Any]:
        """
        Create a new agent from a specification.
        spec may include 'template' key.
        Returns agent info dict.
        """
        return self.generator.create_agent(spec)

    def start_agent(self, agent_id: str) -> bool:
        """
        Start an existing agent (if not already running).
        Loads its config and starts.
        """
        agent_info = self.registry.get_agent(agent_id)
        if not agent_info:
            return False
        config = agent_info.get("spec", {})
        self.lifecycle.start_agent(agent_id, config)
        self.registry.update_status(agent_id, "running")
        return True

    def stop_agent(self, agent_id: str) -> bool:
        """Stop a running agent."""
        if self.lifecycle.stop_agent(agent_id):
            self.registry.update_status(agent_id, "stopped")
            return True
        return False

    def list_agents(self) -> List[Dict[str, Any]]:
        """Return list of all agents (running or not)."""
        return self.registry.list_agents()

    def list_running_agents(self) -> List[str]:
        """Return list of running agent IDs."""
        return self.lifecycle.list_running_agents()

    def delete_agent(self, agent_id: str) -> bool:
        """Delete an agent completely."""
        return self.generator.delete_agent(agent_id)

    def deploy_to_github(self, agent_id: str, repo_name: str = None, private: bool = False) -> str:
        """
        Package an agent and deploy to GitHub.
        Returns the clone URL.
        """
        agent_info = self.registry.get_agent(agent_id)
        if not agent_info:
            raise ValueError(f"Agent {agent_id} not found.")
        # Package the agent (creates a zip or temp folder)
        # For GitHub, we use packager to prepare a folder, then push.
        return self.packager.deploy_github(agent_info, repo_name, private)

    def get_agent_status(self, agent_id: str) -> Optional[str]:
        """Return agent status."""
        info = self.registry.get_agent(agent_id)
        return info.get("status") if info else None

    def send_message_to_agent(self, agent_id: str, message: str, user_id: str = "user") -> str:
        """
        Send a message to a running agent and get a response.
        """
        agent = self.lifecycle.get_agent(agent_id)
        if not agent:
            raise RuntimeError(f"Agent {agent_id} is not running.")
        return agent.process_user_message(user_id, message)
