We'll now implement Step 5: Browser Automation Page in the React UI.

This page provides full control over the browser automation API: launch/close, navigate, screenshot, recording, skills, scheduling, and testing tools. It uses polling to update the screenshot and status.

1. Add browser API functions to src/lib/api.ts

// Browser API functions
export async function browserLaunch(): Promise<{ status: string; active_tab?: string; tabs?: string[] }> {
  return fetchJSON('/browser/launch', { method: 'POST' });
}

export async function browserClose(): Promise<{ status: string }> {
  return fetchJSON('/browser/close', { method: 'POST' });
}

export async function browserNavigate(url: string): Promise<{ url: string; title: string }> {
  return fetchJSON('/browser/navigate', {
    method: 'POST',
    body: JSON.stringify({ url }),
  });
}

export async function browserScreenshot(): Promise<{ image: string }> {
  return fetchJSON('/browser/screenshot');
}

export async function browserState(): Promise<{ running: boolean; url?: string; title?: string }> {
  return fetchJSON('/browser/state');
}

export async function browserRecordStart(): Promise<{ status: string }> {
  return fetchJSON('/browser/record/start', { method: 'POST' });
}

export async function browserRecordStop(): Promise<{ status: string; actions: any[]; file: string }> {
  return fetchJSON('/browser/record/stop', { method: 'POST' });
}

export async function browserRecordList(): Promise<{ skills: string[] }> {
  return fetchJSON('/browser/record/list');
}

export async function browserRecordReplay(skillName: string, state?: any): Promise<any> {
  const query = `?skill_name=${encodeURIComponent(skillName)}`;
  return fetchJSON(`/browser/record/replay${query}`, {
    method: 'POST',
    body: JSON.stringify(state || {}),
  });
}

export async function browserRecordSave(name: string, actions: any[]): Promise<{ status: string; file: string }> {
  return fetchJSON(`/browser/record/save_ai?name=${encodeURIComponent(name)}`, {
    method: 'POST',
    body: JSON.stringify(actions),
  });
}

export async function browserAICreateSkill(description: string, model: string = 'gpt-4o-mini'): Promise<{ actions: any[] }> {
  return fetchJSON(`/browser/record/ai_create?description=${encodeURIComponent(description)}&model=${encodeURIComponent(model)}`, {
    method: 'POST',
  });
}

export async function browserClickTest(): Promise<{ results: any[] }> {
  return fetchJSON('/browser/test/clicks', { method: 'POST' });
}

export async function browserAccessibilityAudit(): Promise<{ violations: any[] }> {
  return fetchJSON('/browser/audit/accessibility');
}

export async function browserPerformance(): Promise<{ metrics: any }> {
  return fetchJSON('/browser/performance');
}

export async function browserNetworkLog(): Promise<{ log: any[] }> {
  return fetchJSON('/browser/network/log');
}

export async function browserTabNew(): Promise<{ status: string; active_tab: string; tabs: string[] }> {
  return fetchJSON('/browser/tab/new', { method: 'POST' });
}

export async function browserTabSwitch(tabId: string): Promise<{ active_tab: string }> {
  return fetchJSON(`/browser/tab/switch?tab_id=${encodeURIComponent(tabId)}`, { method: 'POST' });
}

export async function browserTabClose(tabId: string): Promise<{ active_tab: string; tabs: string[] }> {
  return fetchJSON(`/browser/tab/close?tab_id=${encodeURIComponent(tabId)}`, { method: 'POST' });
}

export async function browserTabList(): Promise<{ tabs: string[]; active: string }> {
  return fetchJSON('/browser/tab/list');
}

export async function browserVisualBaseline(): Promise<{ baseline_saved: string }> {
  return fetchJSON('/browser/visual/baseline', { method: 'POST' });
}

export async function browserVisualCompare(threshold: number = 0.05): Promise<{ mse: number; regression: boolean }> {
  return fetchJSON(`/browser/visual/compare?threshold=${threshold}`, { method: 'POST' });
}

2. Create src/pages/BrowserPage.tsx

import { useState, useEffect, useRef } from 'react';
import {
  browserLaunch,
  browserClose,
  browserNavigate,
  browserScreenshot,
  browserState,
  browserRecordStart,
  browserRecordStop,
  browserRecordList,
  browserRecordReplay,
  browserAICreateSkill,
  browserClickTest,
  browserAccessibilityAudit,
  browserPerformance,
  browserNetworkLog,
  browserTabNew,
  browserTabSwitch,
  browserTabClose,
  browserTabList,
  browserVisualBaseline,
  browserVisualCompare,
} from '../lib/api';

export default function BrowserPage() {
  const [isRunning, setIsRunning] = useState(false);
  const [url, setUrl] = useState('');
  const [screenshot, setScreenshot] = useState('');
  const [tabs, setTabs] = useState<string[]>([]);
  const [activeTab, setActiveTab] = useState('');
  const [skills, setSkills] = useState<string[]>([]);
  const [recording, setRecording] = useState(false);
  const [aiDescription, setAiDescription] = useState('');
  const [resultText, setResultText] = useState('');
  const [pollInterval, setPollInterval] = useState<number | null>(null);

  // Poll browser state and screenshot when running
  useEffect(() => {
    let interval: any;
    const check = async () => {
      try {
        const state = await browserState();
        setIsRunning(state.running);
        if (state.running) {
          const shot = await browserScreenshot();
          setScreenshot(shot.image);
          if (state.url) setUrl(state.url);
          const tabData = await browserTabList();
          setTabs(tabData.tabs);
          setActiveTab(tabData.active);
        }
      } catch (e) {
        // ignore
      }
    };
    if (isRunning) {
      interval = setInterval(check, 2000);
    }
    return () => clearInterval(interval);
  }, [isRunning]);

  const loadSkills = async () => {
    try {
      const data = await browserRecordList();
      setSkills(data.skills);
    } catch (e) {
      console.error(e);
    }
  };

  useEffect(() => {
    loadSkills();
  }, []);

  const handleLaunch = async () => {
    try {
      await browserLaunch();
      setIsRunning(true);
    } catch (e) {
      alert('Launch failed: ' + (e as Error).message);
    }
  };

  const handleClose = async () => {
    try {
      await browserClose();
      setIsRunning(false);
      setScreenshot('');
    } catch (e) {
      alert('Close failed: ' + (e as Error).message);
    }
  };

  const handleNavigate = async () => {
    if (!url) return;
    try {
      await browserNavigate(url);
    } catch (e) {
      alert('Navigate failed: ' + (e as Error).message);
    }
  };

  const handleRecordStart = async () => {
    try {
      await browserRecordStart();
      setRecording(true);
    } catch (e) {
      alert('Record start failed: ' + (e as Error).message);
    }
  };

  const handleRecordStop = async () => {
    try {
      const result = await browserRecordStop();
      setRecording(false);
      setResultText(JSON.stringify(result.actions, null, 2));
      loadSkills();
    } catch (e) {
      alert('Record stop failed: ' + (e as Error).message);
    }
  };

  const handleReplaySkill = async (skillName: string) => {
    try {
      await browserRecordReplay(skillName);
      setResultText(`Replay of ${skillName} completed.`);
    } catch (e) {
      setResultText(`Replay failed: ${(e as Error).message}`);
    }
  };

  const handleAICreateSkill = async () => {
    if (!aiDescription) return;
    try {
      const result = await browserAICreateSkill(aiDescription);
      setResultText(JSON.stringify(result.actions, null, 2));
      // Optional: save automatically? We'll let user decide.
    } catch (e) {
      setResultText(`AI create failed: ${(e as Error).message}`);
    }
  };

  const handleClickTest = async () => {
    try {
      const result = await browserClickTest();
      setResultText(JSON.stringify(result.results, null, 2));
    } catch (e) {
      setResultText(`Click test failed: ${(e as Error).message}`);
    }
  };

  const handleAccessibility = async () => {
    try {
      const result = await browserAccessibilityAudit();
      setResultText(JSON.stringify(result.violations, null, 2));
    } catch (e) {
      setResultText(`Accessibility audit failed: ${(e as Error).message}`);
    }
  };

  const handlePerformance = async () => {
    try {
      const result = await browserPerformance();
      setResultText(JSON.stringify(result.metrics, null, 2));
    } catch (e) {
      setResultText(`Performance test failed: ${(e as Error).message}`);
    }
  };

  const handleNetworkLog = async () => {
    try {
      const result = await browserNetworkLog();
      setResultText(JSON.stringify(result.log, null, 2));
    } catch (e) {
      setResultText(`Network log failed: ${(e as Error).message}`);
    }
  };

  const handleVisualBaseline = async () => {
    try {
      await browserVisualBaseline();
      setResultText('Baseline saved.');
    } catch (e) {
      setResultText(`Baseline failed: ${(e as Error).message}`);
    }
  };

  const handleVisualCompare = async () => {
    try {
      const result = await browserVisualCompare();
      setResultText(JSON.stringify(result, null, 2));
    } catch (e) {
      setResultText(`Compare failed: ${(e as Error).message}`);
    }
  };

  const handleTabNew = async () => {
    try {
      const data = await browserTabNew();
      setTabs(data.tabs);
      setActiveTab(data.active_tab);
    } catch (e) {
      alert('New tab failed: ' + (e as Error).message);
    }
  };

  const handleTabSwitch = async (tabId: string) => {
    try {
      await browserTabSwitch(tabId);
      setActiveTab(tabId);
    } catch (e) {
      alert('Switch tab failed: ' + (e as Error).message);
    }
  };

  const handleTabClose = async (tabId: string) => {
    try {
      const data = await browserTabClose(tabId);
      setTabs(data.tabs);
      setActiveTab(data.active_tab);
    } catch (e) {
      alert('Close tab failed: ' + (e as Error).message);
    }
  };

  return (
    <div className="p-4 flex flex-col h-full">
      <div className="flex flex-wrap items-center gap-2 mb-4">
        <button
          onClick={handleLaunch}
          className="bg-accent text-black px-3 py-1 rounded hover:opacity-90"
          disabled={isRunning}
        >
          Launch
        </button>
        <button
          onClick={handleClose}
          className="bg-red-500/10 text-red-400 px-3 py-1 rounded hover:bg-red-500 hover:text-white"
          disabled={!isRunning}
        >
          Close
        </button>
        <input
          type="text"
          value={url}
          onChange={(e) => setUrl(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && handleNavigate()}
          placeholder="https://example.com"
          className="flex-1 min-w-[200px] bg-surface2 border border-border rounded px-3 py-1 text-sm"
        />
        <button onClick={handleNavigate} className="bg-blue-500/10 text-blue-400 px-3 py-1 rounded hover:bg-blue-500 hover:text-white">
          Go
        </button>
        <button onClick={handleRecordStart} className="bg-green-500/10 text-green-400 px-3 py-1 rounded hover:bg-green-500 hover:text-white">
          {recording ? 'Recording...' : 'Record'}
        </button>
        <button onClick={handleRecordStop} className="bg-yellow-500/10 text-yellow-400 px-3 py-1 rounded hover:bg-yellow-500 hover:text-white">
          Stop
        </button>
      </div>

      {/* Tabs */}
      <div className="flex items-center gap-2 mb-2">
        {tabs.map((tab) => (
          <div
            key={tab}
            className={`px-2 py-1 rounded cursor-pointer text-sm ${
              tab === activeTab ? 'bg-surface2 text-accent' : 'bg-surface1 text-gray-400'
            }`}
            onClick={() => handleTabSwitch(tab)}
          >
            {tab}
            <button
              onClick={(e) => {
                e.stopPropagation();
                handleTabClose(tab);
              }}
              className="ml-1 text-gray-500 hover:text-red-400"
            >
              x
            </button>
          </div>
        ))}
        <button onClick={handleTabNew} className="text-gray-400 hover:text-accent text-sm">
          + New Tab
        </button>
      </div>

      {/* Screenshot area */}
      <div className="flex-1 bg-surface1 border border-border rounded-lg overflow-hidden mb-4">
        {screenshot ? (
          <img src={screenshot} alt="Browser screenshot" className="w-full h-full object-contain" />
        ) : (
          <div className="flex items-center justify-center h-full text-gray-500">
            Browser not running or no screenshot yet.
          </div>
        )}
      </div>

      {/* Testing tools row */}
      <div className="flex flex-wrap gap-2 mb-4">
        <button onClick={handleClickTest} className="bg-purple-500/10 text-purple-300 px-3 py-1 rounded hover:bg-purple-500 hover:text-white">
          Click Test
        </button>
        <button onClick={handleAccessibility} className="bg-purple-500/10 text-purple-300 px-3 py-1 rounded hover:bg-purple-500 hover:text-white">
          Accessibility
        </button>
        <button onClick={handlePerformance} className="bg-purple-500/10 text-purple-300 px-3 py-1 rounded hover:bg-purple-500 hover:text-white">
          Performance
        </button>
        <button onClick={handleNetworkLog} className="bg-purple-500/10 text-purple-300 px-3 py-1 rounded hover:bg-purple-500 hover:text-white">
          Network Log
        </button>
        <button onClick={handleVisualBaseline} className="bg-purple-500/10 text-purple-300 px-3 py-1 rounded hover:bg-purple-500 hover:text-white">
          Visual Baseline
        </button>
        <button onClick={handleVisualCompare} className="bg-purple-500/10 text-purple-300 px-3 py-1 rounded hover:bg-purple-500 hover:text-white">
          Visual Compare
        </button>
      </div>

      {/* Skills and AI creation */}
      <div className="grid grid-cols-2 gap-4 mb-4">
        <div>
          <h3 className="text-sm font-medium mb-2">Saved Skills</h3>
          <div className="max-h-40 overflow-y-auto bg-surface1 border border-border rounded p-2">
            {skills.length === 0 ? (
              <p className="text-gray-500 text-xs">No skills saved yet.</p>
            ) : (
              skills.map((skill) => (
                <div key={skill} className="flex items-center justify-between py-1 border-b border-border last:border-0">
                  <span className="text-sm">{skill}</span>
                  <button
                    onClick={() => handleReplaySkill(skill)}
                    className="text-xs bg-accent/10 text-accent px-2 py-1 rounded hover:bg-accent hover:text-black"
                  >
                    Replay
                  </button>
                </div>
              ))
            )}
          </div>
        </div>
        <div>
          <h3 className="text-sm font-medium mb-2">AI Create Skill</h3>
          <textarea
            value={aiDescription}
            onChange={(e) => setAiDescription(e.target.value)}
            placeholder="Describe the task, e.g., 'Log in to example.com and click Dashboard'"
            className="w-full bg-surface1 border border-border rounded p-2 text-sm mb-2"
            rows={3}
          />
          <button onClick={handleAICreateSkill} className="bg-accent text-black px-3 py-1 rounded hover:opacity-90 text-sm">
            Generate Actions
          </button>
        </div>
      </div>

      {/* Results */}
      {resultText && (
        <div className="bg-surface1 border border-border rounded p-3 mt-2 max-h-60 overflow-auto">
          <pre className="text-xs text-gray-300">{resultText}</pre>
        </div>
      )}
    </div>
  );
}
