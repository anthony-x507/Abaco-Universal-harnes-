📄 install.sh

#!/bin/bash
# install.sh – One-command setup for Aegis Agent Factory (Linux/macOS)

set -e

echo "=== Aegis Agent Factory Setup ==="

# 1. Check Python
if ! command -v python3 &> /dev/null; then
    echo "Python3 is required. Please install Python 3.10+."
    exit 1
fi

# 2. Create virtual environment
if [ ! -d "venv" ]; then
    echo "Creating virtual environment..."
    python3 -m venv venv
fi
source venv/bin/activate

# 3. Upgrade pip and install requirements
echo "Installing dependencies..."
pip install --upgrade pip
pip install -r requirements.txt

# 4. Install Playwright (optional, for browser automation)
if command -v playwright &> /dev/null; then
    echo "Installing Playwright Chromium..."
    playwright install chromium
else
    echo "Playwright not found; skipping browser install. Run 'pip install playwright && playwright install chromium' to enable browser features."
fi

# 5. Copy .env.example if .env missing
if [ ! -f ".env" ]; then
    cp .env.example .env
    echo "Created .env from template. Please edit it to add API keys."
fi

# 6. Create data directories
mkdir -p data/agents data/browser_skills data/browser_state data/uploads logs

# 7. Run tests (optional)
if command -v pytest &> /dev/null; then
    echo "Running integration tests..."
    pytest tests/test_integration.py -q || echo "Tests failed; continuing anyway."
fi

echo ""
echo "Setup complete!"
echo "To start the API and UI, run: ./run.sh"

📄 install.ps1

# install.ps1 – One-command setup for Windows

Write-Host "=== Aegis Agent Factory Setup (Windows) ==="

# 1. Check Python
if (-not (Get-Command python -ErrorAction SilentlyContinue)) {
    Write-Error "Python is required. Install Python 3.10+ and try again."
    exit 1
}

# 2. Create virtual environment
if (-not (Test-Path "venv")) {
    python -m venv venv
}
& .\venv\Scripts\Activate.ps1

# 3. Install requirements
Write-Host "Installing dependencies..."
pip install --upgrade pip
pip install -r requirements.txt

# 4. Copy .env if missing
if (-not (Test-Path ".env")) {
    Copy-Item .env.example .env
    Write-Host "Created .env from template. Edit it to add API keys."
}

# 5. Create directories
New-Item -ItemType Directory -Force -Path data\agents, data\browser_skills, data\browser_state, data\uploads, logs | Out-Null

# 6. Run tests (optional)
if (Get-Command pytest -ErrorAction SilentlyContinue) {
    pytest tests\test_integration.py -q
}

Write-Host "Setup complete!"
Write-Host "To start the API and UI, run: .\run.ps1"

📄 Dockerfile.full

# Dockerfile.full – Full container for Aegis Agent Factory
FROM python:3.10-slim

WORKDIR /app

# Install system dependencies (for SDR, etc., if needed)
RUN apt-get update && apt-get install -y --no-install-recommends \
    git \
    curl \
    && rm -rf /var/lib/apt/lists/*

# Copy requirements and install
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Install Playwright (optional)
RUN pip install playwright && playwright install chromium

# Copy project
COPY . .

# Create directories
RUN mkdir -p data/agents data/browser_skills data/browser_state data/uploads logs

EXPOSE 8000

CMD ["uvicorn", "api.main:app", "--host", "0.0.0.0", "--port", "8000"]
