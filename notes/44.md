// Scheduler API functions
export interface ScheduledTask {
  skill_name: string;
  run_at?: string;
  cron?: string;
  timezone?: string;
  repeat_interval_minutes?: number;
  status: string;
  last_run?: string;
  error?: string;
  next_run?: string;
}

export async function listScheduledTasks(): Promise<{ tasks: ScheduledTask[] }> {
  return fetchJSON('/scheduler/list');
}

export async function addScheduledTask(task: {
  skill_name: string;
  run_at?: string;
  cron?: string;
  timezone?: string;
  repeat_interval_minutes?: number;
}): Promise<{ status: string; task: ScheduledTask }> {
  return fetchJSON('/scheduler/add', {
    method: 'POST',
    body: JSON.stringify(task),
  });
}

export async function deleteScheduledTask(index: number): Promise<{ removed: ScheduledTask }> {
  return fetchJSON(`/scheduler/${index}`, { method: 'DELETE' });
}

export async function addNaturalLanguageSchedule(description: string, model: string = 'gpt-4o-mini'): Promise<{
  status: string;
  task: ScheduledTask;
  parsed: any;
}> {
  return fetchJSON('/scheduler/nl_add', {
    method: 'POST',
    body: JSON.stringify({ description, model }),
  });
}

2. Create src/pages/SchedulerPage.tsx

import { useState, useEffect } from 'react';
import {
  listScheduledTasks,
  addScheduledTask,
  deleteScheduledTask,
  addNaturalLanguageSchedule,
  ScheduledTask,
} from '../lib/api';

export default function SchedulerPage() {
  const [tasks, setTasks] = useState<ScheduledTask[]>([]);
  const [skillName, setSkillName] = useState('');
  const [runAt, setRunAt] = useState('');
  const [cron, setCron] = useState('');
  const [timezone, setTimezone] = useState('UTC');
  const [nlDescription, setNlDescription] = useState('');
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadTasks();
  }, []);

  const loadTasks = async () => {
    setLoading(true);
    try {
      const data = await listScheduledTasks();
      setTasks(data.tasks);
    } catch (e) {
      console.error(e);
    } finally {
      setLoading(false);
    }
  };

  const handleAddManual = async () => {
    if (!skillName || (!runAt && !cron)) {
      alert('Please provide skill name and either run_at or cron');
      return;
    }
    try {
      await addScheduledTask({
        skill_name: skillName,
        run_at: runAt || undefined,
        cron: cron || undefined,
        timezone,
      });
      loadTasks();
      // Reset fields
      setSkillName('');
      setRunAt('');
      setCron('');
    } catch (e) {
      alert('Failed to schedule: ' + (e as Error).message);
    }
  };

  const handleAddNL = async () => {
    if (!nlDescription) return;
    try {
      const result = await addNaturalLanguageSchedule(nlDescription);
      alert(`Scheduled: ${JSON.stringify(result.parsed)}`);
      setNlDescription('');
      loadTasks();
    } catch (e) {
      alert('Natural language scheduling failed: ' + (e as Error).message);
    }
  };

  const handleDelete = async (index: number) => {
    try {
      await deleteScheduledTask(index);
      loadTasks();
    } catch (e) {
      alert('Delete failed: ' + (e as Error).message);
    }
  };

  return (
    <div className="p-4">
      <h1 className="text-xl font-semibold mb-4">Scheduler</h1>

      <div className="grid grid-cols-2 gap-4">
        {/* Manual scheduling */}
        <div className="bg-surface1 border border-border rounded-lg p-4">
          <h2 className="text-lg font-medium mb-3">Manual Schedule</h2>
          <div className="space-y-3">
            <input
              type="text"
              value={skillName}
              onChange={(e) => setSkillName(e.target.value)}
              placeholder="Skill name"
              className="w-full bg-surface2 border border-border rounded px-3 py-2 text-sm"
            />
            <div className="grid grid-cols-2 gap-2">
              <input
                type="datetime-local"
                value={runAt}
                onChange={(e) => setRunAt(e.target.value)}
                className="bg-surface2 border border-border rounded px-3 py-2 text-sm"
              />
              <input
                type="text"
                value={cron}
                onChange={(e) => setCron(e.target.value)}
                placeholder="Cron (e.g., 0 9 * * *)"
                className="bg-surface2 border border-border rounded px-3 py-2 text-sm"
              />
            </div>
            <input
              type="text"
              value={timezone}
              onChange={(e) => setTimezone(e.target.value)}
              placeholder="Timezone"
              className="w-full bg-surface2 border border-border rounded px-3 py-2 text-sm"
            />
            <button
              onClick={handleAddManual}
              className="bg-accent text-black font-medium rounded-lg px-4 py-2 hover:opacity-90"
            >
              Schedule
            </button>
          </div>
        </div>

        {/* Natural language scheduling */}
        <div className="bg-surface1 border border-border rounded-lg p-4">
          <h2 className="text-lg font-medium mb-3">AI Schedule (Natural Language)</h2>
          <textarea
            value={nlDescription}
            onChange={(e) => setNlDescription(e.target.value)}
            placeholder="e.g., Run login skill every day at 9am"
            className="w-full bg-surface2 border border-border rounded p-3 text-sm mb-3"
            rows={4}
          />
          <button
            onClick={handleAddNL}
            className="bg-purple-500/10 text-purple-300 px-4 py-2 rounded hover:bg-purple-500 hover:text-white"
          >
            Schedule with AI
          </button>
        </div>
      </div>

      {/* Task list */}
      <div className="mt-6">
        <h2 className="text-lg font-medium mb-3">Scheduled Tasks</h2>
        <div className="bg-surface1 border border-border rounded-lg p-2">
          {loading ? (
            <p className="text-gray-500">Loading...</p>
          ) : tasks.length === 0 ? (
            <p className="text-gray-500">No tasks scheduled.</p>
          ) : (
            tasks.map((task, index) => (
              <div
                key={index}
                className="flex items-center justify-between py-2 px-2 border-b border-border last:border-0"
              >
                <div>
                  <span className="font-medium">{task.skill_name}</span>
                  <div className="text-xs text-gray-400">
                    {task.cron ? `Cron: ${task.cron}` : `At: ${task.run_at || task.next_run}`}
                    {' · '}
                    {task.status}
                    {task.error && ` · Error: ${task.error}`}
                  </div>
                </div>
                <button
                  onClick={() => handleDelete(index)}
                  className="text-xs bg-red-500/10 text-red-400 px-2 py-1 rounded hover:bg-red-500 hover:text-white"
                >
                  Delete
                </button>
              </div>
            ))
          )}
        </div>
      </div>
    </div>
  );
}
