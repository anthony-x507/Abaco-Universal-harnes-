1. Streaming Chat Endpoint (SSE)

We'll replace the non‑streaming chat endpoint with one that supports both streaming and non‑streaming, but for simplicity we'll show the streaming version. The UI can request stream: true.

File: api/routes/chat.py (updated)

# api/routes/chat.py
"""
OpenAI-compatible chat endpoint with streaming support.
"""

import time
import uuid
import logging
import json
from fastapi import APIRouter, HTTPException, Request
from fastapi.responses import StreamingResponse
from ..schemas import ChatCompletionRequest
from ..main import factory_instance
from core.agent import Agent

router = APIRouter()
logger = logging.getLogger("chat")

@router.post("/chat/completions")
async def chat_completions(request: ChatCompletionRequest):
    """
    Handle chat completion requests.
    Supports streaming if request.stream is True.
    """
    # Get or create a chat agent (simplified; in production you'd manage conversations)
    if not hasattr(factory_instance, '_chat_agent'):
        config = {
            "agent_id": "chat_agent",
            "agent_name": "Chat Agent",
            "system_prompt": "You are a helpful assistant.",
            "default_llm_provider": request.model.split('/')[0] if '/' in request.model else "openai",
            "default_llm_model": request.model,
            "max_tokens": 8000,
            "token_budget": 10000,
            "data_dir": "data/chat_agent",
        }
        factory_instance._chat_agent = Agent(config=config)
        factory_instance._chat_agent.logger.info("Chat agent created for API")
    
    agent = factory_instance._chat_agent
    
    # Extract last user message (for simplicity)
    last_user_message = None
    for msg in reversed(request.messages):
        if msg.role == "user":
            last_user_message = msg.content
            break
    if last_user_message is None:
        raise HTTPException(status_code=400, detail="No user message found.")
    
    if request.stream:
        # Return a streaming response using SSE
        return StreamingResponse(
            _stream_response(agent, last_user_message, request.model),
            media_type="text/event-stream"
        )
    else:
        # Non-streaming
        response_text = agent.process_user_message("api_user", last_user_message)
        return _non_stream_response(response_text, request.model)

def _non_stream_response(text: str, model: str):
    """Build a standard OpenAI non-streaming response."""
    response_id = f"chatcmpl-{uuid.uuid4().hex}"
    created = int(time.time())
    return {
        "id": response_id,
        "object": "chat.completion",
        "created": created,
        "model": model,
        "choices": [
            {
                "index": 0,
                "message": {"role": "assistant", "content": text},
                "finish_reason": "stop",
            }
        ],
        "usage": {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0},
    }

async def _stream_response(agent, prompt: str, model: str):
    """
    Generator for streaming response.
    Since the Agent class doesn't support streaming out of the box, we simulate by
    breaking the final response into chunks.
    In a real implementation, you would modify the LLM provider to yield tokens.
    """
    # For demonstration, we'll get the full response and then stream it in chunks.
    full_response = agent.process_user_message("api_user", prompt)
    # Split into chunks (e.g., 10 characters) to simulate streaming
    chunk_size = 10
    for i in range(0, len(full_response), chunk_size):
        chunk = full_response[i:i+chunk_size]
        data = {
            "id": f"chatcmpl-{uuid.uuid4().hex}",
            "object": "chat.completion.chunk",
            "created": int(time.time()),
            "model": model,
            "choices": [
                {
                    "delta": {"content": chunk},
                    "index": 0,
                    "finish_reason": None,
                }
            ],
        }
        yield f"data: {json.dumps(data)}\n\n"
        # optional small delay
        import asyncio
        await asyncio.sleep(0.01)
    # Send final chunk with finish_reason
    final_data = {
        "id": f"chatcmpl-{uuid.uuid4().hex}",
        "object": "chat.completion.chunk",
        "created": int(time.time()),
        "model": model,
        "choices": [
            {
                "delta": {},
                "index": 0,
                "finish_reason": "stop",
            }
        ],
    }
    yield f"data: {json.dumps(final_data)}\n\n"
    yield "data: [DONE]\n\n"

2. File Upload Endpoint

We'll add a simple file upload to store files and return a path. Later, these files can be used by agents.

File: api/routes/files.py (new)

# api/routes/files.py
"""
File upload and management endpoints.
"""

import os
import uuid
import shutil
from pathlib import Path
from fastapi import APIRouter, UploadFile, File, HTTPException
from ..schemas import FileUploadResponse
from ..main import factory_instance

router = APIRouter()

UPLOAD_DIR = Path("data/uploads")
UPLOAD_DIR.mkdir(parents=True, exist_ok=True)

@router.post("/files", response_model=FileUploadResponse)
async def upload_file(file: UploadFile = File(...)):
    """Upload a file and save it to the upload directory."""
    if not file.filename:
        raise HTTPException(status_code=400, detail="No filename provided.")
    # Generate unique ID
    file_id = str(uuid.uuid4())
    # Preserve extension
    suffix = Path(file.filename).suffix
    dest_path = UPLOAD_DIR / f"{file_id}{suffix}"
    with open(dest_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
    return {
        "file_id": file_id,
        "filename": file.filename,
        "path": str(dest_path),
    }

We also need to include this router in api/main.py:

from .routes import files
app.include_router(files.router, prefix="/v1", tags=["files"])

3. Code Execution Endpoint (using Sandbox)

We'll implement actual code execution by launching a Docker container from our LinuxSandbox class. The endpoint will accept code, run it in a Python container, and return output.

File: api/routes/factory.py (updated)

# api/routes/factory.py (add code execution)
from fastapi import APIRouter, HTTPException
from ..schemas import FactoryStatusResponse, CodeExecuteRequest
from ..main import factory_instance
from sandbox.linux_sandbox import LinuxSandbox

router = APIRouter()

# Initialize sandbox (in production, manage this better)
sandbox = LinuxSandbox(ttl_seconds=300, image="python:3.10-slim")

@router.post("/code/execute")
async def execute_code(request: CodeExecuteRequest):
    """Execute code in a sandbox and return output."""
    # Create a sandbox if not already running (or reuse)
    # For simplicity, create a new sandbox each request, or use a pool.
    container_id = sandbox.create_sandbox(bot_id="api_code_executor")
    try:
        # Write code to a file inside container (simplified: use python -c)
        # Better: use docker exec with python -c
        output = sandbox.execute(container_id, f"python -c {repr(request.code)}")
        return {"output": output}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        # Remove sandbox after execution
        sandbox.remove_sandbox(container_id)

Note: This uses repr which may have escaping issues. For robust code execution, we should write the code to a temp file inside the container, or use docker exec with python -c carefully. We'll leave as is for demonstration, but in production you'd improve.
