# api/routes/browser.py
"""
Browser automation endpoints using Playwright.
Provides launch, navigate, click, record, and screenshot capabilities.
"""

import asyncio
import base64
import json
import logging
from pathlib import Path
from typing import Optional, Dict, Any, List
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

router = APIRouter()
logger = logging.getLogger("browser")

# Playwright instance (lazy)
_browser = None
_page = None
_recording = False
_recorded_actions = []

RECORDINGS_DIR = Path("data/browser_recordings")
RECORDINGS_DIR.mkdir(parents=True, exist_ok=True)

class BrowserAction(BaseModel):
    url: str = None
    selector: str = None
    x: int = None
    y: int = None
    text: str = None

@router.post("/launch")
async def launch_browser():
    """Launch a headless Chromium browser."""
    global _browser, _page
    if _browser:
        return {"status": "already_running"}
    try:
        from playwright.async_api import async_playwright
        playwright = await async_playwright().start()
        _browser = await playwright.chromium.launch(headless=True)
        _page = await _browser.new_page()
        return {"status": "launched"}
    except Exception as e:
        logger.error(f"Browser launch failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/close")
async def close_browser():
    global _browser, _page
    if _browser:
        await _browser.close()
        _browser = None
        _page = None
    return {"status": "closed"}

@router.post("/navigate")
async def navigate(action: BrowserAction):
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    if not action.url:
        raise HTTPException(status_code=400, detail="URL required")
    await _page.goto(action.url)
    return {"url": _page.url, "title": await _page.title()}

@router.post("/click")
async def click(action: BrowserAction):
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    if action.selector:
        await _page.click(action.selector)
    elif action.x and action.y:
        await _page.mouse.click(action.x, action.y)
    else:
        raise HTTPException(status_code=400, detail="Selector or coordinates required")
    return {"status": "clicked"}

@router.get("/screenshot")
async def screenshot():
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    shot = await _page.screenshot()
    b64 = base64.b64encode(shot).decode('utf-8')
    return {"image": f"data:image/png;base64,{b64}"}

@router.post("/record/start")
async def record_start():
    global _recording, _recorded_actions
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    _recording = True
    _recorded_actions = []
    # Inject a script to listen for clicks and inputs
    await _page.evaluate("""
        window.__recordedActions = [];
        document.addEventListener('click', (e) => {
            if (window.__recordedActions) {
                window.__recordedActions.push({type:'click', selector: e.target.tagName, text: e.target.innerText?.slice(0,100)});
            }
        });
        document.addEventListener('input', (e) => {
            if (window.__recordedActions && e.target.tagName === 'INPUT') {
                window.__recordedActions.push({type:'input', selector: e.target.name || e.target.id, value: e.target.value});
            }
        });
    """)
    return {"status": "recording_started"}

@router.post("/record/stop")
async def record_stop():
    global _recording, _recorded_actions
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    if not _recording:
        return {"status": "not_recording"}
    # Fetch recorded actions from page
    actions = await _page.evaluate("window.__recordedActions || []")
    _recorded_actions = actions
    _recording = False
    # Save to file
    filename = RECORDINGS_DIR / f"recording_{int(__import__('time').time())}.json"
    with open(filename, 'w') as f:
        json.dump(actions, f, indent=2)
    return {"status": "recording_stopped", "actions": actions, "file": str(filename)}

@router.post("/record/play")
async def record_play(file_path: str):
    if not _page:
        raise HTTPException(status_code=400, detail="Browser not launched")
    with open(file_path, 'r') as f:
        actions = json.load(f)
    for action in actions:
        if action.get('type') == 'click':
            # We need selector; we stored tagName and text, not sufficient; this is a placeholder
            pass
    return {"status": "played", "actions_count": len(actions)}

@router.get("/state")
async def browser_state():
    if not _page:
        return {"running": False}
    return {"running": True, "url": _page.url, "title": await _page.title()}

// Browser functions
async function launchBrowser() {
    try {
        await fetchJSON(`${API_BASE}/browser/launch`, { method: 'POST' });
        updateBrowserView();
    } catch (e) { alert('Launch failed: ' + e.message); }
}
async function closeBrowser() {
    try {
        await fetchJSON(`${API_BASE}/browser/close`, { method: 'POST' });
        document.getElementById('browser-view').innerHTML = '<p>Browser closed.</p>';
    } catch (e) { alert('Close failed: ' + e.message); }
}
async function navigateBrowser() {
    const url = document.getElementById('browser-url').value;
    if (!url) return;
    try {
        await fetchJSON(`${API_BASE}/browser/navigate`, {
            method: 'POST',
            body: JSON.stringify({ url })
        });
        updateBrowserView();
    } catch (e) { alert('Navigate failed: ' + e.message); }
}
async function updateBrowserView() {
    try {
        const state = await fetchJSON(`${API_BASE}/browser/state`);
        if (state.running) {
            const shot = await fetchJSON(`${API_BASE}/browser/screenshot`);
            document.getElementById('browser-view').innerHTML = `<img src="${shot.image}" />`;
            document.getElementById('browser-url').value = state.url;
        } else {
            document.getElementById('browser-view').innerHTML = '<p>Browser not running.</p>';
        }
    } catch (e) { /* ignore */ }
}
async function recordStart() {
    try {
        await fetchJSON(`${API_BASE}/browser/record/start`, { method: 'POST' });
        alert('Recording started. Interact with the browser.');
    } catch (e) { alert('Record start failed: ' + e.message); }
}
async function recordStop() {
    try {
        const result = await fetchJSON(`${API_BASE}/browser/record/stop`, { method: 'POST' });
        alert('Recording saved: ' + result.file);
    } catch (e) { alert('Record stop failed: ' + e.message); }
}
// Event listeners
document.getElementById('browser-launch-btn').addEventListener('click', launchBrowser);
document.getElementById('browser-close-btn').addEventListener('click', closeBrowser);
document.getElementById('browser-go-btn').addEventListener('click', navigateBrowser);
document.getElementById('browser-screenshot-btn').addEventListener('click', updateBrowserView);
document.getElementById('browser-record-start-btn').addEventListener('click', recordStart);
document.getElementById('browser-record-stop-btn').addEventListener('click', recordStop);
// Poll browser view every 2 seconds if browser is running
setInterval(() => {
    fetch(`${API_BASE}/browser/state`).then(res => res.json()).then(state => {
        if (state.running) updateBrowserView();
    }).catch(() => {});
}, 2000);
