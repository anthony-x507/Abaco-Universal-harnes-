2. Update tools/browser_scheduler.py to support cron and timezone

Updated tools/browser_scheduler.py

# tools/browser_scheduler.py (updated)
import threading
import time
import requests
import json
from datetime import datetime, timedelta
from typing import List, Dict, Any, Optional
from croniter import croniter
import pytz  # add pytz to requirements

class BrowserScheduler:
    def __init__(self, api_base_url="http://localhost:8000"):
        self.api_base_url = api_base_url
        self.tasks: List[Dict[str, Any]] = []
        self.lock = threading.Lock()
        self.thread = threading.Thread(target=self._run_loop, daemon=True)
        self.thread.start()

    def add_task(self, skill_name: str, run_at: str = None, cron: str = None,
                 timezone: str = "UTC", repeat_interval_minutes: int = 0) -> Dict:
        """
        Add a task with either run_at (ISO datetime) or cron expression.
        """
        if not run_at and not cron:
            raise ValueError("Either run_at or cron must be provided")
        task = {
            "skill_name": skill_name,
            "run_at": run_at,
            "cron": cron,
            "timezone": timezone,
            "repeat_interval_minutes": repeat_interval_minutes,
            "status": "scheduled",
            "last_run": None,
            "error": None,
            "next_run": run_at if run_at else self._compute_next_cron(cron, timezone),
        }
        with self.lock:
            self.tasks.append(task)
        return task

    def _compute_next_cron(self, cron_expr: str, timezone: str) -> str:
        """Compute next run time based on cron expression and timezone."""
        tz = pytz.timezone(timezone)
        now = datetime.now(tz)
        cron = croniter(cron_expr, now)
        next_dt = cron.get_next(datetime)
        return next_dt.isoformat()

    def get_tasks(self) -> List[Dict]:
        with self.lock:
            return list(self.tasks)

    def delete_task(self, index: int) -> Optional[Dict]:
        with self.lock:
            if 0 <= index < len(self.tasks):
                return self.tasks.pop(index)
        return None

    def _run_loop(self):
        while True:
            now = datetime.utcnow()
            with self.lock:
                for task in self.tasks:
                    if task["status"] not in ("scheduled", "retry"):
                        continue
                    run_at = datetime.fromisoformat(task["next_run"])
                    if now >= run_at:
                        self._execute_task(task, now)
            time.sleep(10)

    def _execute_task(self, task: Dict, now: datetime):
        try:
            response = requests.post(
                f"{self.api_base_url}/v1/browser/record/replay",
                params={"skill_name": task["skill_name"]},
                timeout=60,
            )
            if response.status_code == 200:
                task["status"] = "executed"
                task["last_run"] = now.isoformat()
                task["error"] = None
            else:
                task["status"] = "error"
                task["error"] = f"Replay failed with status {response.status_code}"
        except Exception as e:
            task["status"] = "error"
            task["error"] = str(e)

        # Determine next run
        if task["cron"]:
            # Recompute next cron time
            task["next_run"] = self._compute_next_cron(task["cron"], task["timezone"])
            task["status"] = "scheduled"
        elif task["repeat_interval_minutes"] > 0:
            next_run = now + timedelta(minutes=task["repeat_interval_minutes"])
            task["next_run"] = next_run.isoformat()
            task["status"] = "scheduled"
        else:
            task["status"] = "completed"

Modify ScheduleRequest model and add_schedule endpoint.

# api/routes/scheduler.py (updated)
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import Optional
from tools.browser_scheduler import BrowserScheduler

router = APIRouter()
scheduler = BrowserScheduler()

class ScheduleRequest(BaseModel):
    skill_name: str
    run_at: Optional[str] = None
    cron: Optional[str] = None
    timezone: str = "UTC"
    repeat_interval_minutes: int = 0

@router.post("/scheduler/add")
async def add_schedule(request: ScheduleRequest):
    try:
        task = scheduler.add_task(
            skill_name=request.skill_name,
            run_at=request.run_at,
            cron=request.cron,
            timezone=request.timezone,
            repeat_interval_minutes=request.repeat_interval_minutes,
        )
        return {"status": "scheduled", "task": task}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.get("/scheduler/list")
async def list_schedules():
    return {"tasks": scheduler.get_tasks()}

@router.delete("/scheduler/{task_index}")
async def delete_schedule(task_index: int):
    removed = scheduler.delete_task(task_index)
    if removed is None:
        raise HTTPException(status_code=404, detail="Task not found")
    return {"removed": removed}
