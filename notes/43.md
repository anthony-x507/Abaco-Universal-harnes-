1. Add plugin API functions to src/lib/api.ts

export async function listPlugins(): Promise<{ plugins: string[] }> {
  return fetchJSON('/plugins');
}

export async function executePlugin(pluginName: string, action: string, parameters: Record<string, any>): Promise<{ result: any }> {
  return fetchJSON('/plugins/execute', {
    method: 'POST',
    body: JSON.stringify({ plugin_name: pluginName, action, parameters }),
  });
}

2. Create src/pages/PluginsPage.tsx

import { useState, useEffect } from 'react';
import { listPlugins, executePlugin } from '../lib/api';

export default function PluginsPage() {
  const [plugins, setPlugins] = useState<string[]>([]);
  const [selectedPlugin, setSelectedPlugin] = useState('');
  const [action, setAction] = useState('');
  const [parameters, setParameters] = useState('{}');
  const [result, setResult] = useState('');
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadPlugins();
  }, []);

  const loadPlugins = async () => {
    setLoading(true);
    try {
      const data = await listPlugins();
      setPlugins(data.plugins);
      if (data.plugins.length > 0) setSelectedPlugin(data.plugins[0]);
    } catch (e) {
      console.error(e);
    } finally {
      setLoading(false);
    }
  };

  const handleExecute = async () => {
    if (!selectedPlugin || !action) {
      alert('Please select a plugin and enter an action');
      return;
    }
    let params;
    try {
      params = JSON.parse(parameters);
    } catch (e) {
      alert('Invalid JSON parameters');
      return;
    }
    setLoading(true);
    try {
      const response = await executePlugin(selectedPlugin, action, params);
      setResult(JSON.stringify(response.result, null, 2));
    } catch (e) {
      setResult(`Error: ${(e as Error).message}`);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="p-4">
      <h1 className="text-xl font-semibold mb-4">Plugins</h1>

      <div className="grid grid-cols-2 gap-4">
        {/* Plugin list */}
        <div>
          <h2 className="text-lg font-medium mb-2">Available Plugins</h2>
          <div className="bg-surface1 border border-border rounded-lg p-2 max-h-[60vh] overflow-y-auto">
            {loading ? (
              <p className="text-gray-500">Loading...</p>
            ) : plugins.length === 0 ? (
              <p className="text-gray-500">No plugins loaded.</p>
            ) : (
              plugins.map((plugin) => (
                <div
                  key={plugin}
                  onClick={() => setSelectedPlugin(plugin)}
                  className={`p-2 rounded cursor-pointer ${
                    selectedPlugin === plugin ? 'bg-surface2 text-accent' : 'hover:bg-surface2'
                  }`}
                >
                  {plugin}
                </div>
              ))
            )}
          </div>
        </div>

        {/* Plugin execution */}
        <div>
          <h2 className="text-lg font-medium mb-2">Execute Plugin Action</h2>
          <div className="bg-surface1 border border-border rounded-lg p-4 space-y-3">
            <div>
              <label className="block text-sm text-gray-400">Selected Plugin</label>
              <input
                type="text"
                value={selectedPlugin}
                onChange={(e) => setSelectedPlugin(e.target.value)}
                className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
                readOnly
              />
            </div>
            <div>
              <label className="block text-sm text-gray-400">Action</label>
              <input
                type="text"
                value={action}
                onChange={(e) => setAction(e.target.value)}
                placeholder="e.g., search"
                className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm"
              />
            </div>
            <div>
              <label className="block text-sm text-gray-400">Parameters (JSON)</label>
              <textarea
                value={parameters}
                onChange={(e) => setParameters(e.target.value)}
                rows={5}
                className="w-full bg-surface2 border border-border rounded px-2 py-1 text-sm font-mono"
              />
            </div>
            <button
              onClick={handleExecute}
              disabled={loading}
              className="bg-accent text-black font-medium rounded-lg px-4 py-2 hover:opacity-90"
            >
              Execute
            </button>
            {result && (
              <pre className="bg-surface2 border border-border rounded p-2 text-xs text-gray-300 max-h-60 overflow-auto">
                {result}
              </pre>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
