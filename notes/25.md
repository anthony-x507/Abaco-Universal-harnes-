2. Visual Regression Testing

We'll add endpoints to capture a screenshot of the current page as a baseline and later compare with a new screenshot using pixel difference (via Pillow). If difference exceeds threshold, flag as regression.

from PIL import Image
import io
import numpy as np

@router.post("/visual/baseline")
async def visual_baseline():
    page = get_active_page()
    if not page:
        raise HTTPException(status_code=400, detail="No active tab")
    shot = await page.screenshot()
    # Save baseline
    baseline_path = Path("data/browser_baselines") / "baseline.png"
    baseline_path.parent.mkdir(parents=True, exist_ok=True)
    with open(baseline_path, 'wb') as f:
        f.write(shot)
    return {"baseline_saved": str(baseline_path)}

@router.post("/visual/compare")
async def visual_compare(threshold: float = 0.05):
    page = get_active_page()
    if not page:
        raise HTTPException(status_code=400, detail="No active tab")
    current_shot = await page.screenshot()
    baseline_path = Path("data/browser_baselines") / "baseline.png"
    if not baseline_path.exists():
        raise HTTPException(status_code=400, detail="No baseline found. Save baseline first.")
    # Compare using Pillow
    baseline_img = Image.open(baseline_path).convert('RGB')
    current_img = Image.open(io.BytesIO(current_shot)).convert('RGB')
    # Resize current to baseline size (they should be same dimensions)
    if baseline_img.size != current_img.size:
        # If different, scale current to baseline
        current_img = current_img.resize(baseline_img.size)
    diff = np.abs(np.asarray(baseline_img, dtype=np.int16) - np.asarray(current_img, dtype=np.int16))
    mse = np.mean(diff)
    # mse > threshold indicates significant change
    regression = mse > threshold
    return {"mse": float(mse), "regression": bool(regression)}
